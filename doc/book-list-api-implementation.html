书籍分页列表 API

POST/v1/book/list

管理员按页读取全部书籍。每条书籍带有所属天体，响应同时返回当前页、每页条数、总页数和书籍总数。

校验页大小，安全计算偏移和总页数

将分页查询逻辑放在项目根目录的 paginate/paginate.go。请求和响应中的字段定义分别放在 param/req/pagination 和 param/resp/pagination。用 paginate.NewPage(currentPage, itemPerPage) 构造一次分页条件。按第 2 页、每页 2 条计算，OFFSET = (2 - 1) * 2 = 2，数据库跳过前两条。先用除法判断乘法是否会溢出，可避免极大页码回绕到第一页。最大页大小集中在 MaxPageSize。

Page.Paginate 的签名符合 GORM Scope，模型使用 Scopes(page.Paginate)。计数查询独立执行，能统计分页前的完整结果。GORM 分页 Scope

新增完整文件paginate/paginate.go
package paginate

import (
	"errors"
	"fmt"
	"math"

	"gorm.io/gorm"
)

// MaxPageSize 限制单次查询返回的最大记录数.
const MaxPageSize = 100

// ErrInvalidPage 表示分页参数缺失, 越界或偏移量不能表示为 int.
var ErrInvalidPage = errors.New("分页参数不合法")

// Page 保存校验后的 LIMIT 和 OFFSET, 通过 NewPage 构造.
type Page struct {
	size   int // size 是本页允许返回的最大条数.
	offset int // offset 是跳过的记录数.
}

// NewPage 校验分页范围, 并在乘法之前检查偏移是否会溢出.
func NewPage(currentPage, itemPerPage int) (Page, error) {
	if currentPage < 1 {
		return Page{}, fmt.Errorf("%w: 页码必须从 1 开始", ErrInvalidPage)
	}
	if itemPerPage < 1 || itemPerPage > MaxPageSize {
		return Page{}, fmt.Errorf("%w: 每页条数须在 1 到 %d 之间", ErrInvalidPage, MaxPageSize)
	}
	if currentPage-1 > math.MaxInt/itemPerPage {
		return Page{}, fmt.Errorf("%w: 页码过大", ErrInvalidPage)
	}
	return Page{size: itemPerPage, offset: (currentPage - 1) * itemPerPage}, nil
}

// Valid 判断 Page 是否由合法参数构造, 可用于拦截零值.
func (p Page) Valid() bool {
	return p.size >= 1 && p.size <= MaxPageSize && p.offset >= 0
}

// Offset 返回分页起点, 模型可以据此判断是否还有记录.
func (p Page) Offset() int {
	return p.offset
}

// Paginate 为已排序的 GORM 查询应用分页范围, 可传入 Scopes.
func (p Page) Paginate(db *gorm.DB) *gorm.DB {
	return db.Limit(p.size).Offset(p.offset)
}

// TotalPages 计算总页数, 只使用整数并避免 total + size - 1 溢出.
func (p Page) TotalPages(total int64) int64 {
	if total <= 0 || !p.Valid() {
		return 0
	}
	return 1 + (total-1)/int64(p.size)
}

在同一快照中统计、取页和加载天体

在现有 model/book.go 的 import 分组中加入下方三个导入，再追加三个 *model.Book 方法。业务层只调用 ListPage(ctx, page)。分页条件、总数和行切片都保存在本次调用中，接收者自身的 ID 和字段不参与筛选。

  1. 统计书籍。 ListPage() 开启只读事务，对 book 执行 COUNT(*)。空表或偏移达到总数时跳过后续查询，提交事务后返回空数组和总数。
  2. 读取当前页。 readPage(tx, page) 将 ORDER BY planet_id ASC, sort ASC, id ASC 和分页 Scope 交给数据库执行。最后的书籍主键使排序具有确定的顺序。
  3. 补齐这一页的天体。 loadPlanets(tx, records) 收集本页不同的 planet_id，用一次 IN 查询加载关联，再按原书籍顺序回填。现有 model.Book.Planet 带有 gorm:"-"，需要显式赋值。

MySQL InnoDB 的只读 REPEATABLE READ 事务让三次普通读取看到同一个快照。并发创建的书不会让同一个响应出现总数与条目不一致。每次翻页都是新的请求，期间新增或删除记录仍可能改变后续页面的边界。MySQL InnoDB 一致性读取

现有表通过 ID 关联天体。若本页某条书籍引用的天体已不存在，模型返回带书籍和天体 ID 的错误，业务层包装成 DatabaseError。调用方收到统一的 10201，便于发现数据问题，也能避免访问空的天体指针。

// ListPage 在一个只读事务中统计和读取书籍, 返回当前页以及分页前的总条数.
func (b *Book) ListPage(ctx context.Context, page paginate.Page) ([]*Book, int64, error) {
	if b == nil || ctx == nil {
		return nil, 0, errors.New("nil book model or list context")
	}
	if !page.Valid() {
		return nil, 0, errors.New("invalid book list pagination")
	}
	if db.DB == nil {
		return nil, 0, errors.New("database is not initialized")
	}

	tx := db.DB.WithContext(ctx).Begin(&sql.TxOptions{
		Isolation: sql.LevelRepeatableRead,
		ReadOnly:  true,
	})
	if tx.Error != nil {
		return nil, 0, tx.Error
	}
	// 提前返回或 panic 时回滚. 成功提交后, 延迟回滚不会改变已提交的结果.
	defer tx.Rollback()

	var total int64
	if err := tx.Model(&Book{}).Count(&total).Error; err != nil {
		return nil, 0, err
	}
	records := make([]*Book, 0)
	if int64(page.Offset()) < total {
		var err error
		records, err = b.readPage(tx, page)
		if err != nil {
			return nil, 0, err
		}
	}
	if err := tx.Commit().Error; err != nil {
		return nil, 0, err
	}
	return records, total, nil
}

// readPage 按天体 ID, 排序值和书籍 ID 读取一页, 再补齐所属天体.
func (b *Book) readPage(tx *gorm.DB, page paginate.Page) ([]*Book, error) {
	var records []*Book
	err := tx.Model(&Book{}).
		Select("id", "planet_id", "title", "sort", "display_status").
		Order("planet_id ASC").
		Order("sort ASC").
		Order("id ASC").
		Scopes(page.Paginate).
		Find(&records).Error
	if err != nil {
		return nil, err
	}
	if err := b.loadPlanets(tx, records); err != nil {
		return nil, err
	}
	return records, nil
}

// loadPlanets 用同一个事务批量查询本页涉及的天体, 保留书籍的原顺序.
func (b *Book) loadPlanets(tx *gorm.DB, records []*Book) error {
	if len(records) == 0 {
		return nil
	}
	planetIDs := make([]int64, 0, len(records))
	seenIDs := make(map[int64]struct{}, len(records))
	for _, record := range records {
		if record == nil {
			return errors.New("book list data is inconsistent: nil book record")
		}
		if _, exists := seenIDs[record.PlanetID]; exists {
			continue
		}
		seenIDs[record.PlanetID] = struct{}{}
		planetIDs = append(planetIDs, record.PlanetID)
	}

	planets := make([]*Planet, 0, len(planetIDs))
	if err := tx.Model(&Planet{}).
		Select("id", "name", "title", "intro").
		Where("id IN ?", planetIDs).
		Find(&planets).Error; err != nil {
		return err
	}

	planetsByID := make(map[int64]*Planet, len(planets))
	for _, planet := range planets {
		if planet == nil {
			return errors.New("book list data is inconsistent: nil planet record")
		}
		planetsByID[planet.ID] = planet
	}
	for _, record := range records {
		planet, exists := planetsByID[record.PlanetID]
		if !exists {
			return fmt.Errorf("book list data is inconsistent: book %d references missing planet %d", record.ID, record.PlanetID)
		}
		record.Planet = planet
	}
	return nil
}

// List 获取一页书籍, 返回列表, 总条数和总页数.
func (b *Book) List(ctx context.Context, req *reqBook.ListReq) ([]*Book, int64, int64, error) {
	if b == nil || ctx == nil {
		return nil, 0, 0, &sys_error.InternalError{Cause: errors.New("nil book entity or context")}
	}
	if req == nil || req.Pagination == nil || req.Pagination.CurrentPage == nil || req.Pagination.ItemPerPage == nil {
		return nil, 0, 0, paginate.ErrInvalidPage
	}
	input := req.Pagination
	page, err := paginate.NewPage(*input.CurrentPage, *input.ItemPerPage)
	if err != nil {
		return nil, 0, 0, err
	}

	orm := &model.Book{}
	records, total, err := orm.ListPage(ctx, page)
	if err != nil {
		return nil, 0, 0, &sys_error.DatabaseError{Cause: err}
	}

	books := make([]*Book, 0, len(records))
	for _, record := range records {
		item := &Book{}
		item.fillByORM(ctx, record)
		books = append(books, item)
	}
	return books, total, page.TotalPages(total), nil
}

// List 本函数用于绑定分页请求, 调用业务层并组装书籍列表响应.
func List(c *gin.Context) {
	req := &reqBook.ListReq{}
	respBody := &resp_body.RespBody{}
	if err := c.ShouldBindJSON(req); err != nil {
		respBody.ValidateError(err)
		c.JSON(http.StatusOK, respBody)
		return
	}

	bookBiz := &biz.Book{}
	books, totalItem, totalPage, err := bookBiz.List(c.Request.Context(), req)
	if err != nil {
		var dbErr *sys_error.DatabaseError
		switch {
		case errors.Is(err, paginate.ErrInvalidPage):
			respBody.ValidateError(err)
		case errors.As(err, &dbErr):
			respBody.DBError(dbErr)
		default:
			respBody.InternalError(err)
		}
		c.JSON(http.StatusOK, respBody)
		return
	}

	// 显式初始化非 nil 切片, 确保空页输出 [] 而非 null.
	listResp := &respBook.ListResp{
		Books: make([]*respBook.ListBookResp, 0, len(books)),
		Pagination: &respPagination.Pagination{
			CurrentPage: *req.Pagination.CurrentPage,
			ItemPerPage: *req.Pagination.ItemPerPage,
			TotalPage:   totalPage,
			TotalItem:   totalItem,
		},
	}
	for _, item := range books {
		// 两种 ID 均转为字符串, 其余字段通过响应 DTO 显式选择.
		bookResp := &respBook.ListBookResp{
			ID:            strconv.FormatInt(item.ID, 10),
			Title:         item.Title,
			Sort:          item.Sort,
			DisplayStatus: item.DisplayStatus,
			Planet: &respPlanet.ListPlanetResp{
				ID:    strconv.FormatInt(item.Planet.ID, 10),
				Name:  item.Planet.Name,
				Title: item.Planet.Title,
				Intro: item.Planet.Intro,
			},
		}
		listResp.Books = append(listResp.Books, bookResp)
	}

	respBody.Success(map[string]any{
		"books":      listResp.Books,
		"pagination": listResp.Pagination,
	})
	c.JSON(http.StatusOK, respBody)
}

验收场景应观察到的结果
空表，请求第 1 页业务码 200，books: []，totalItem: 0，totalPage: 0。
同一个天体有多本书，数据跨两页按天体数值 ID、书籍 sort、书籍 ID 升序。隐藏书和同名书均返回。静态数据集翻页无漏项、无重项。
总数是页大小的整倍数或末页不足例如 4 本、每页 2 条，总页数为 2；3 本、每页 2 条，第 2 页仅 1 本。两页的总数相同。
合法页超过末页；每页 100 条前者成功并回显请求页码、真实总数与空数组；后者是允许的上限。
缺少 pagination 或字段；null、0、负数、小数、字符串；每页 101 条或偏移溢出返回 10101 和 data: {}，无分页查询。
书籍或天体 ID 超过 JavaScript 安全整数上限两个 ID 仍是原始数字的十进制字符串，天体名称、标题和简介完整。
有效请求但令牌缺失、无效或过期；数据库查询失败分别走现有 20004、20003 或 10201，错误响应 data: {}。
