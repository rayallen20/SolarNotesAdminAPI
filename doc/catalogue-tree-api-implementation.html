数据库事实对实现的影响
book.sort 是 INT UNSIGNED NOT NULL；catalogue.sort 是 INT UNSIGNED DEFAULT NULL。书籍排序使用 uint32；目录模型、业务实体和节点响应 DTO 均使用 *uint32。根指向 0，已排序子节点的范围为 1 至 4294967295，待排序子节点为 nil。
根目录 ID 为 2123684178040987648，parent_id = 0，type = folder。数据库用整数 0 标识根，序列化后 parent_id 是字符串 "0"。
全书 59 个节点：1 个可见根文件夹、20 个隐藏子文件夹、38 个隐藏文件。58 个非根节点均待排序，其中有 2 个空文件夹。遍历响应应得到 59 个节点。当前样本的编排进度为 0 / 58，前端只统计非根节点。
文件节点名称实际包含 0. 大致过程.md。article.catalogue_id 上有唯一索引。原样返回 catalogue.name。文件类型取自 catalogue.type，无需拼接扩展名或联查 article.path。
已生效的 chk_catalogue_sort_by_parent 检查父节点与排序值。根节点的排序值必须为 0；非根节点允许 NULL 或正数。组树时仍校验待排序节点的显隐和根节点类型，这两个条件由业务层负责。
现有唯一索引为 uk_book_parent_sort(book_id, parent_id, sort)，表使用 InnoDB。按书籍查目录可使用索引左前缀。多条待排记录可共存；父节点属于同一本书、树无环等关系仍须在组装时确认。

此前只读核对这本书时，目录共有 1 个根、21 个文件夹（含根）和 38 个文件，合计 59 个节点。从根算第 1 层，最深为第 7 层。

程序侧的三个排序字段

将 model.Catalogue.Sort、biz.Catalogue.Sort 和 TreeNodeResp.Sort 统一声明为 *uint32。原模型的有符号 32 位整数在读取 2147483648 起的值时会报错。指针保留数据库 NULL；响应标签继续使用 json:"sort"，这样待排序节点会显式返回 "sort": null。

目录树请求只接收 book.id；成功响应仍用 JSON 数字传送排序值，JavaScript 的 number 可以精确表示全部 uint32 值。书籍摘要中的 TreeBookResp.Sort 对应另一个非空列，继续使用 uint32。

业务层的入口和返回值

建议入口　(*biz.Catalogue).Tree(ctx, req) (*biz.Book, []*biz.Catalogue, error)

调用顺序　控制器绑定请求 → 模型读取书籍和目录行 → 业务层挂接、校验、排序 → 控制器转换响应 DTO。

biz.Book 和 biz.Catalogue 是独立的业务实体。一次成功调用返回书籍摘要和该书的根节点切片。即使目录行数为零，第一个返回值仍是查询到的书籍，第二个是空切片。若书籍不存在，则两个数据返回值均为 nil，同时返回书籍不存在错误。

数据库层用 catalogue.book_id 确定归属，业务节点只保存 Parent 和 Children 两种节点间关系。根的 Parent == nil，每个子节点的 Parent 都指向其直接父节点。书籍摘要直接从本次快照映射。已有 biz.Book.fillByORM 面向同时加载天体的记录，读树只需要书籍自身的四个字段。biz 处理树的有效性和顺序；控制器负责把数值 ID 转成字符串、构造响应对象并调用 RespBody.Success。

在同一快照读取书籍与目录行

沿用 model.Book.OptionsByPlanetID 的只读可重复读事务。先按主键取书籍摘要，再用 WHERE book_id = ? 取该书全部目录。两次查询共用 tx，使并发导入或删除时读到的书籍与目录属于同一快照。记录读完即提交，再到内存组树。查询包含 intro，用于填充现有业务实体的 Intro；接口③的节点 DTO 仅输出约定的展示字段。

package model

import (
	"context"
	"database/sql"
	"errors"

	"SolarNotesAdminAPI/db"

	"gorm.io/gorm"
)

// Catalogue 映射读树使用的 catalogue 表字段.
type Catalogue struct {
	ID            int64   `gorm:"column:id;primaryKey;autoIncrement:false"` // ID 为目录节点主键.
	BookID        int64   `gorm:"column:book_id"`                           // BookID 为所属书籍.
	ParentID      int64   `gorm:"column:parent_id"`                         // ParentID 为直接父节点 ID, 根节点为 0.
	Name          string  `gorm:"column:name"`                              // Name 为节点显示名.
	Intro         string  `gorm:"column:intro"`                             // Intro 为节点介绍.
	Type          string  `gorm:"column:type"`                              // Type 为 folder 或 file.
	Sort          *uint32 `gorm:"column:sort"`                              // Sort 对应 INT UNSIGNED, nil 表示待排序.
	DisplayStatus string  `gorm:"column:display_status"`                    // DisplayStatus 为节点自身的显隐状态.
}

// ReadBookTree 在同一个只读快照内取书籍摘要和该书全部目录行.
func (c *Catalogue) ReadBookTree(ctx context.Context, bookID int64) (*Book, []Catalogue, error) {
	if c == nil {
		return nil, nil, errors.New("model.Catalogue.ReadBookTree: nil receiver")
	}
	if ctx == nil {
		return nil, nil, errors.New("model.Catalogue.ReadBookTree: nil context")
	}
	if bookID <= 0 {
		return nil, nil, errors.New("model.Catalogue.ReadBookTree: book ID must be positive")
	}
	if db.DB == nil {
		return nil, nil, errors.New("model.Catalogue.ReadBookTree: database is not initialized")
	}

	tx := db.DB.WithContext(ctx).Begin(&sql.TxOptions{
		Isolation: sql.LevelRepeatableRead,
		ReadOnly:  true,
	})
	if tx.Error != nil {
		return nil, nil, tx.Error
	}
	defer tx.Rollback()

	var book Book
	err := tx.Select("id", "title", "sort", "display_status").
		Where("id = ?", bookID).Take(&book).Error
	if err != nil {
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return nil, nil, BookNotFoundErr
		}
		return nil, nil, err
	}

	var rows []Catalogue
	err = tx.Model(&Catalogue{}).
		Select("id", "book_id", "parent_id", "name", "intro", "type", "sort", "display_status").
		Where("book_id = ?", bookID).Find(&rows).Error
	if err != nil {
		return nil, nil, err
	}
	if err := tx.Commit().Error; err != nil {
		return nil, nil, err
	}
	return &book, rows, nil
}

第一条 Take 未命中时返回 model.BookNotFoundErr。第二条 Find 未命中时得到空集合。循环查询每个目录的子项会使 SQL 次数随节点数增加；此实现读取已存在书籍的 SQL 查询次数固定为两次。

// Tree 根据请求读取一本书的当前状态并组装完整的管理目录.
func (c *Catalogue) Tree(ctx context.Context, req *reqCatalogue.TreeReq) (*Book, []*Catalogue, error) {
	if c == nil {
		return nil, nil, &sys_error.InternalError{Cause: errors.New("biz.Catalogue.Tree: nil receiver")}
	}
	if ctx == nil {
		return nil, nil, &sys_error.InternalError{Cause: errors.New("biz.Catalogue.Tree: nil request context")}
	}
	if req == nil {
		return nil, nil, &sys_error.InternalError{Cause: errors.New("biz.Catalogue.Tree: nil tree request")}
	}
	if req.Book == nil {
		return nil, nil, &sys_error.InternalError{Cause: errors.New("biz.Catalogue.Tree: nil book reference")}
	}

	orm := &model.Catalogue{}
	book, rows, err := orm.ReadBookTree(ctx, req.Book.ID)
	if err != nil {
		if errors.Is(err, model.BookNotFoundErr) {
			return nil, nil, sys_error.BookNotFoundErr
		}
		return nil, nil, &sys_error.DatabaseError{Cause: err}
	}
	roots, err := buildCatalogueTree(book.ID, rows)
	if err != nil {
		return nil, nil, &sys_error.InternalError{Cause: err}
	}
	summary := &Book{
		ID:            book.ID,
		Title:         book.Title,
		Sort:          book.Sort,
		DisplayStatus: book.DisplayStatus,
	}
	return summary, roots, nil
}

两遍挂接，再核对可达节点数

第一遍按 ID 建立索引并定位根；第二遍根据行中的 ParentID 找到直接父文件夹，把 child.Parent 指向父节点，同时向 parent.Children 追加同一个子节点指针。即使输入中子行早于父行也能正确组装。随后从根遍历，每个非根节点只有一个父项；若遍历数量小于原始行数，说明有与根不连通的环形分支。非空数据要求一个 folder 根。根允许显隐切换，非根的已排项可隐藏，待排项只能隐藏。

package biz

import (
	"fmt"

	"SolarNotesAdminAPI/enum"
	"SolarNotesAdminAPI/model"

	"golang.org/x/text/collate"
	"golang.org/x/text/language"
)

// buildCatalogueTree 将单本书的平面记录挂接为双向树, 空书返回空数组.
func buildCatalogueTree(bookID int64, rows []model.Catalogue) ([]*Catalogue, error) {
	if len(rows) == 0 {
		return []*Catalogue{}, nil
	}

	byID := make(map[int64]*Catalogue, len(rows))
	var root *Catalogue
	for i := range rows {
		row := &rows[i]
		if row.BookID != bookID {
			return nil, fmt.Errorf("node %d belongs to book %d, expected %d", row.ID, row.BookID, bookID)
		}
		if _, exists := byID[row.ID]; exists {
			return nil, fmt.Errorf("duplicate catalogue ID %d", row.ID)
		}
		if err := validateCatalogueRow(row); err != nil {
			return nil, err
		}

		node := &Catalogue{
			ID:            row.ID,
			Type:          row.Type,
			Name:          row.Name,
			Intro:         row.Intro,
			Sort:          row.Sort,
			DisplayStatus: row.DisplayStatus,
			Children:      []*Catalogue{},
		}
		if row.ParentID == 0 {
			if root != nil {
				return nil, fmt.Errorf("multiple roots in book %d", bookID)
			}
			root = node
		}
		byID[node.ID] = node
	}
	if root == nil {
		return nil, fmt.Errorf("root missing in nonempty book %d", bookID)
	}

	// 按模型行的 ParentID 挂接两端, 根节点的 Parent 保持 nil.
	for _, row := range rows {
		if row.ParentID == 0 {
			continue
		}
		parent := byID[row.ParentID]
		if parent == nil || parent.Type != enum.TypeFolder {
			return nil, fmt.Errorf("node %d has missing or non-folder parent %d", row.ID, row.ParentID)
		}
		node := byID[row.ID]
		node.Parent = parent
		parent.Children = append(parent.Children, node)
	}

	// 只沿 Children 从根向下遍历. 每个非根节点恰有一个父节点, 不可达分支必然含环.
	order := catalogueOrder{names: collate.New(language.Make("zh-CN"), collate.Numeric)}
	queue := []*Catalogue{root}
	for i := 0; i < len(queue); i++ {
		node := queue[i]
		if err := order.sortChildren(node); err != nil {
			return nil, err
		}
		queue = append(queue, node.Children...)
	}
	if len(queue) != len(rows) {
		return nil, fmt.Errorf("cycle outside the root of book %d", bookID)
	}
	return []*Catalogue{root}, nil
}

// validateCatalogueRow 在挂接父节点前, 按数据库 ParentID 区分根和普通节点并校验字段.
func validateCatalogueRow(row *model.Catalogue) error {
	if row.ID <= 0 || row.ParentID < 0 {
		return fmt.Errorf("invalid catalogue identity: id=%d parent_id=%d", row.ID, row.ParentID)
	}
	if row.Type != enum.TypeFolder && row.Type != enum.TypeFile {
		return fmt.Errorf("node %d has unknown type %q", row.ID, row.Type)
	}
	if row.DisplayStatus != enum.DisplayVisible && row.DisplayStatus != enum.DisplayHidden {
		return fmt.Errorf("node %d has unknown display status %q", row.ID, row.DisplayStatus)
	}
	if row.ParentID == 0 {
		if row.Type != enum.TypeFolder || row.Sort == nil || *row.Sort != 0 {
			return fmt.Errorf("root %d must be a folder with sort 0", row.ID)
		}
		return nil
	}
	if row.Sort == nil && row.DisplayStatus != enum.DisplayHidden {
		return fmt.Errorf("unsorted node %d must be hidden", row.ID)
	}
	if row.Sort != nil && *row.Sort == 0 {
		return fmt.Errorf("non-root node %d has invalid sort %d", row.ID, *row.Sort)
	}
	return nil
}

每个文件夹独立排序

已排项全部在前，按 sort 数值升序；待排项随后，按中文名称自然顺序排列，名称比较相等时按数值 ID 升序。比如同层的已排「结语」仍排在待排「第2章」「第10章」前。不同父目录可以各自拥有 sort = 1。

原型的名称比较使用 Intl.Collator('zh-CN', { numeric: true })。Go 示例用 golang.org/x/text/collate 的中文规则和 collate.Numeric 实现相同的排序意图。不同排序库版本的少数字符可能有差异，页面直接使用服务端的 children 顺序。工程目前已包含 golang.org/x/text v0.34.0，落地后运行 go mod tidy 可将其归为直接依赖。

package biz

import (
	"cmp"
	"fmt"
	"slices"

	"golang.org/x/text/collate"
)

// catalogueOrder 持有本次组树使用的中文排序器.
type catalogueOrder struct {
	names *collate.Collator // names 让名称中的连续数字按数值进行比较.
}

// sortChildren 排列指定节点的直接子项, 并检查同一父项下是否有重复序号.
func (o catalogueOrder) sortChildren(parent *Catalogue) error {
	children := parent.Children
	slices.SortFunc(children, o.compare)
	for i := 1; i < len(children); i++ {
		previous, current := children[i-1].Sort, children[i].Sort
		if previous != nil && current != nil && *previous == *current {
			return fmt.Errorf("folder %d has duplicate child sort %d", parent.ID, *current)
		}
	}
	return nil
}

// compare 将已排序项放在前面, 待排序项使用中文自然序, 最后以数值 ID 兜底.
func (o catalogueOrder) compare(a, b *Catalogue) int {
	switch {
	case a.Sort != nil && b.Sort == nil:
		return -1
	case a.Sort == nil && b.Sort != nil:
		return 1
	case a.Sort != nil && b.Sort != nil:
		if order := cmp.Compare(*a.Sort, *b.Sort); order != 0 {
			return order
		}
	default:
		if order := o.names.CompareString(a.Name, b.Name); order != 0 {
			return order
		}
	}
	return cmp.Compare(a.ID, b.ID)
}

设全书有 n 个节点，索引、挂接和可达性检查均为 O(n)；同级分别排序，最坏比较次数为 O(n log n)。业务层的索引、节点与遍历队列合计使用 O(n) 空间。

控制器将两个成功返回值组装到同一个 data.book 对象。转换时只沿 Children 向下递归；访问 Parent 时仅读取其 ID。这样既能生成 parent_id，也能避免沿双向引用重复遍历。

package catalogue

import (
	"encoding/json"
	"errors"
	"io"
	"net/http"
	"strconv"

	"SolarNotesAdminAPI/biz"
	reqCatalogue "SolarNotesAdminAPI/param/req/v1/catalogue"
	respCatalogue "SolarNotesAdminAPI/param/resp/v1/catalogue"
	"SolarNotesAdminAPI/resp_body"
	"SolarNotesAdminAPI/sys_error"

	"github.com/gin-gonic/gin"
	"github.com/gin-gonic/gin/binding"
)

// Tree 返回选中书籍的摘要和完整目录.
func Tree(c *gin.Context) {
	req := &reqCatalogue.TreeReq{}
	body := &resp_body.RespBody{}
	if err := bindTreeRequest(c, req); err != nil {
		body.ValidateError(err)
		c.JSON(http.StatusOK, body)
		return
	}

	service := &biz.Catalogue{}
	book, roots, err := service.Tree(c.Request.Context(), req)
	if err != nil {
		var dbErr *sys_error.DatabaseError
		switch {
		case errors.Is(err, sys_error.BookNotFoundErr):
			body.BookNotFoundError()
		case errors.As(err, &dbErr):
			body.DBError(dbErr)
		default:
			body.InternalError(err)
		}
		c.JSON(http.StatusOK, body)
		return
	}

	bookResp := &respCatalogue.TreeBookResp{
		ID:            strconv.FormatInt(book.ID, 10),
		Title:         book.Title,
		Sort:          book.Sort,
		DisplayStatus: book.DisplayStatus,
		Catalogues:    treeNodeResponses(roots),
	}
	body.Success(map[string]any{"book": bookResp})
	c.JSON(http.StatusOK, body)
}

// bindTreeRequest 只接受一个完整 JSON 值, 然后执行 Gin 的字段校验.
func bindTreeRequest(c *gin.Context, req *reqCatalogue.TreeReq) error {
	decoder := json.NewDecoder(c.Request.Body)
	if err := decoder.Decode(req); err != nil {
		return err
	}
	if err := decoder.Decode(&struct{}{}); err != io.EOF {
		return errors.New("request body must contain one JSON object")
	}
	return binding.Validator.ValidateStruct(req)
}

// treeNodeResponses 保留业务层的顺序, 并把每层 ID 转为字符串.
func treeNodeResponses(nodes []*biz.Catalogue) []*respCatalogue.TreeNodeResp {
	result := make([]*respCatalogue.TreeNodeResp, 0, len(nodes))
	for _, node := range nodes {
		parentID := "0"
		if node.Parent != nil {
			parentID = strconv.FormatInt(node.Parent.ID, 10)
		}
		result = append(result, &respCatalogue.TreeNodeResp{
			ID:            strconv.FormatInt(node.ID, 10),
			ParentID:      parentID,
			Name:          node.Name,
			Type:          node.Type,
			Sort:          node.Sort,
			DisplayStatus: node.DisplayStatus,
			Children:      treeNodeResponses(node.Children),
		})
	}
	return result
}
