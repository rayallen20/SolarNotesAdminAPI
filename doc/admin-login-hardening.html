02 / CONFIGURATION

在启动时验证 JWT 有效期

ExpireSeconds 必须大于 0，并且转换为 time.Duration 纳秒时不能溢出。下面在原有的签发方、受众和密钥检查之外，增加秒数边界；仍通过 conf.Load() 的既有调用链触发。

当前开发环境的一周配置保持 604800 秒。最大可转换值用于防止数值溢出，不代表建议把 JWT 设置为那么长的有效期；生产环境需要配置有效的 JWT 参数和密钥。

替换完整文件conf/jwt.go
package conf

import (
	"encoding/base64"
	"errors"
	"strings"
	"time"
)

// JWT 保存签发方、受众、有效期及用于验签的密钥配置。
type JWT struct {
	Issuer        string `yaml:"issuer"`         // Issuer JWT的ISS,标识允许签发令牌的后端
	Audience      string `yaml:"audience"`       // Audience JWT的AUD,标识令牌预期服务的前端
	ExpireSeconds int64  `yaml:"expire_seconds"` // ExpireSeconds JWT的过期时间,单位秒
	Secret        string `yaml:"secret"`         // Secret JWT的签名密钥
	signingKey    []byte // signingKey 从 Secret 解码出的原始字节(私有字段不参与YAML绑定)
}

// validate 本方法用于验证JWT配置的合法性,若合法则将秘钥保存至 signingKey 字段中,若不合法则返回错误并阻止配置加载
func (j *JWT) validate() error {
	j.Issuer = strings.TrimSpace(j.Issuer)
	j.Audience = strings.TrimSpace(j.Audience)

	if j.Issuer == "" || j.Audience == "" {
		return errors.New("JWT配置错误: Issuer和Audience不能为空字符串")
	}

	// time.Duration 使用有符号 64 位纳秒，先检查秒数再转换，避免溢出。
	const maxExpireSeconds int64 = (1<<63 - 1) / int64(time.Second)
	if j.ExpireSeconds <= 0 || j.ExpireSeconds > maxExpireSeconds {
		return errors.New("JWT配置错误: expire_seconds 必须是有效的正秒数")
	}

	encoded := strings.TrimSpace(j.Secret)
	if encoded == "" {
		return errors.New("JWT配置错误: Secret不能为空字符串")
	}

	key, err := base64.StdEncoding.DecodeString(encoded)
	if err != nil {
		return errors.New("JWT配置错误: Secret必须是Base64编码的字符串")
	}

	if len(key) < 32 {
		return errors.New("JWT配置错误: Secret解码后的字节长度必须至少为32字节")
	}

	// 复制字节切片以避免外部修改
	j.signingKey = append([]byte(nil), key...)

	return nil
}

// SigningKey 本方法用于获取JWT的签名密钥,为防止调用方修改签名密钥,返回的字节切片为签名密钥的副本
func (j *JWT) SigningKey() []byte {
	return append([]byte(nil), j.signingKey...)
}

// Lifetime 本方法用于获取JWT的有效期,单位为time.Duration
func (j *JWT) Lifetime() time.Duration {
	return time.Duration(j.ExpireSeconds) * time.Second
}

package conf

import (
	"bytes"
	"encoding/base64"
	"testing"
	"time"
)

// TestJWTExpiryValidation 覆盖零、负数、一周、最大值和溢出边界。
func TestJWTExpiryValidation(t *testing.T) {
	const maxSeconds int64 = (1<<63 - 1) / int64(time.Second)
	cases := []struct {
		name    string // name 是场景名称。
		seconds int64  // seconds 是测试的有效秒数。
		valid   bool   // valid 表示此秒数是否应该通过校验。
	}{
		{"zero", 0, false},
		{"negative", -1, false},
		{"one week", 604800, true},
		{"maximum", maxSeconds, true},
		{"overflow", maxSeconds + 1, false},
	}
	for _, tc := range cases {
		j := &JWT{
			Issuer:        "test-issuer",
			Audience:      "test-audience",
			ExpireSeconds: tc.seconds,
			Secret:        base64.StdEncoding.EncodeToString(bytes.Repeat([]byte{7}, 32)),
		}
		err := j.validate()
		if (err == nil) != tc.valid {
			t.Errorf("%s: valid=%v, error=%v", tc.name, tc.valid, err)
		}
		if err == nil && j.Lifetime() <= 0 {
			t.Errorf("%s: lifetime must stay positive", tc.name)
		}
	}
}
03 / RATE LIMIT

为登录路由增加频率限制

示例按 c.ClientIP() 计数：同一 IP 在滚动 60 秒内最多放行 5 个登录请求。无论请求 JSON 是否有效、登录是否成功，只要到达登录路由就计入；当前全局 CORS 中间件会先结束 OPTIONS 预检。

第 6 个请求由中间件直接返回，不进入 Controller、Biz 或数据库。沿用项目的 HTTP 200 与业务码约定，新增 20005；前端应将它作为“稍后再试”处理，不应当当作令牌过期。

JSON 示例登录请求触发限流 · HTTP 200
{
  "code": 20005,
  "message": "请求过于频繁，请稍后再试",
  "data": {}
}

这是单进程、按 IP 的示例：重启会清空计数，共用出口 IP 的用户会共享额度；多个实例各自计数，分布式来源也能分别计数。多实例应改用共享限流存储或网关，并结合账号与网络层策略。示例最多保留 10,000 个客户端，达到上限时暂时拒绝新客户端，直到过期记录被清理。

新增限流中间件

限流次数与窗口是示例常量，可根据访问规模调整；可信代理地址在下一节通过 YAML 配置。记录在请求进入时清理，不启动后台 goroutine；互斥锁保护并发计数。

新增完整文件middleware/login_rate_limiter.go
package middleware

import (
	"net/http"
	"sync"
	"time"

	"SolarNotesAdminAPI/resp_body"
	"github.com/gin-gonic/gin"
)

const (
	// loginWindow 是统计登录请求次数的滚动时间窗口。
	loginWindow = time.Minute
	// loginMaxRequests 是单个客户端在窗口内允许的最大请求数。
	loginMaxRequests = 5
	// loginMaxClients 限制内存中同时跟踪的客户端数量。
	loginMaxClients = 10000
)

// LoginRateLimiter 在单个进程内按客户端 IP 限制登录请求。
type LoginRateLimiter struct {
	// mu 保护 requests 和 lastCleanup，允许多个请求并发访问。
	mu sync.Mutex
	// requests 为每个客户端保存当前窗口内已放行请求的时间。
	requests map[string][]time.Time
	// lastCleanup 记录上次移除过期客户端记录的时间。
	lastCleanup time.Time
}

// NewLoginRateLimiter 创建可供登录路由共享的限流器实例。
func NewLoginRateLimiter() *LoginRateLimiter {
	return &LoginRateLimiter{requests: make(map[string][]time.Time)}
}

// Handle 是 Gin 中间件；超过限制时返回统一业务响应并停止后续处理。
func (l *LoginRateLimiter) Handle(c *gin.Context) {
	c.Header("Cache-Control", "no-store")
	if !l.allow(c.ClientIP()) {
		body := &resp_body.RespBody{}
		body.LoginRateLimit()
		c.AbortWithStatusJSON(http.StatusOK, body)
		return
	}
	c.Next()
}

// allow 判断客户端是否可以发起一次登录请求；成功时记录该次请求。
func (l *LoginRateLimiter) allow(clientIP string) bool {
	l.mu.Lock()
	defer l.mu.Unlock()

	// 取时间也放在锁内，确保同一 IP 的记录按时间顺序追加。
	now := time.Now()
	cutoff := now.Add(-loginWindow)
	if l.lastCleanup.IsZero() || now.Sub(l.lastCleanup) >= loginWindow {
		for ip, history := range l.requests {
			if len(history) == 0 {
				delete(l.requests, ip)
				continue
			}
			lastRequestTime := history[len(history)-1]
			if !lastRequestTime.After(cutoff) {
				delete(l.requests, ip)
			}
		}
		l.lastCleanup = now
	}

	history := l.requests[clientIP]
	first := 0
	for first < len(history) && !history[first].After(cutoff) {
		first++
	}
	history = history[first:]
	if len(history) >= loginMaxRequests {
		return false
	}

	// 到达容量时拒绝新客户端，等待过期记录清理；已有客户端仍按窗口判断。
	if _, exists := l.requests[clientIP]; !exists && len(l.requests) >= loginMaxClients {
		return false
	}
	l.requests[clientIP] = append(history, now)
	return true
}

扩展响应码并固定内部错误文案

下面的完整响应文件新增 LoginRateLimitCode 和 LoginRateLimit()。同时落实第 05 节的对外错误策略：DBError(err) 与 InternalError(err) 的调用形式保留，但响应不再包含任意底层错误文本。

替换完整文件resp_body/resp_body.go
package resp_body

// 响应码规则:
// 200: 成功响应
// 500: 服务端内部错误
//
// 10XXX: 系统级别错误
//
//	101XX: 参数校验错误
//	102XX: 数据库错误
//
// 20XXX: 具体业务错误
const (
	SuccessCode              = 200 // SuccessCode 成功响应状态码
	ServiceInternalErrorCode = 500 // ServiceInternalErrorCode 服务端内部错误状态码

	ValidateErrorCode = 10101 // ValidateErrorCode 参数校验错误状态码
	DBErrorCode       = 10201 // DBErrorCode 数据库错误状态码

	LoginCredentialErrorCode = 20001 // LoginCredentialErrorCode 账号或密码错误状态码
	LoginStateChangedCode    = 20002 // LoginStateChangedCode 登录期间凭据状态变化,需重新登录状态码
	TokenExpiredCode         = 20003 // TokenExpiredCode token过期状态码
	InvalidTokenCode         = 20004 // InvalidTokenCode token无效状态码
	LoginRateLimitCode       = 20005 // LoginRateLimitCode 登录请求触发访问频率限制
)

// RespBody 是项目统一的 JSON 响应体。
type RespBody struct {
	Code    int            `json:"code"`    // Code 响应体中的状态码部分
	Message string         `json:"message"` // Message 响应体中的消息部分
	Data    map[string]any `json:"data"`    // Data 响应体中的数据部分 即有效载荷
}

// Success 本方法用于设置响应成功响应体
func (r *RespBody) Success(data map[string]any) {
	r.Code = SuccessCode
	r.Message = "success"
	r.Data = data
}

// failure 本方法用于统一设置错误业务码/消息/空对象 Data ,避免输出 null 或遗留成功数据
func (r *RespBody) failure(code int, message string) {
	r.Code = code
	r.Message = message
	r.Data = map[string]any{}
}

// ValidateError 本方法用于设置参数校验错误时的响应体
func (r *RespBody) ValidateError(err error) {
	message := "请求参数不合法"
	if err != nil {
		message = err.Error()
	}
	r.failure(ValidateErrorCode, message)
}

// DBError 本方法用于设置数据库错误时的响应体
func (r *RespBody) DBError(_ error) {
	r.failure(DBErrorCode, "数据库操作失败，请稍后重试")
}

// InternalError 本方法用于生成统一内部错误响应体,不向前端输出底层错误
func (r *RespBody) InternalError(_ error) {
	r.failure(ServiceInternalErrorCode, "服务暂时不可用，请稍后重试")
}

// LoginRateLimit 返回登录请求过于频繁时的业务码和统一消息。
func (r *RespBody) LoginRateLimit() {
	r.failure(LoginRateLimitCode, "请求过于频繁，请稍后再试")
}

// LoginCredentialError 本方法用于设置账号不存在/密码不正确时的响应体,不向前端输出具体错误
func (r *RespBody) LoginCredentialError() {
	r.failure(LoginCredentialErrorCode, "账号或密码错误")
}

// LoginStateChangedError 本方法用于设置验证用户名密码后,管理员记录的条件更新未命中时的响应体,不向前端输出具体错误
func (r *RespBody) LoginStateChangedError() {
	r.failure(LoginStateChangedCode, "账号状态已变化,请重新登录")
}

// TokenExpiredError 本方法用于设置token过期时的响应体
func (r *RespBody) TokenExpiredError() {
	r.failure(TokenExpiredCode, "登录已过期,请重新登录")
}

// InvalidToken 本方法用于生成令牌无效或已撤销响应时的响应体
func (r *RespBody) InvalidToken() {
	r.failure(InvalidTokenCode, "登录状态无效,请重新登录")
}

在服务配置中声明可信代理

在现有 conf.Service 中增加代理地址列表。Config 已经持有 Service，无需新增配置对象；配置加载器启用了严格字段检查，因此应先增加结构体字段，再在 YAML 中填写。

替换完整文件conf/service.go
package conf

// Service 服务相关配置。
type Service struct {
	Port           string   `yaml:"port"`            // Port 服务监听端口。
	TrustedProxies []string `yaml:"trusted_proxies"` // TrustedProxies 可信代理 IP 或 CIDR 列表，空列表表示不信任代理。
}

以下 YAML 仅展示需要合并到已有 service 段的配置；保留原有 port 和其他配置，不要再创建一个同名的 service 段。示例配置文件与无代理的本地开发环境使用空列表：

合并配置片段config.example.yaml / config.dev.yaml · 直接连接 Go 服务
service:
  # 保留现有的 port 等配置
  trusted_proxies: []

生产环境如果采用本文的同机 Nginx 示例，并且 Nginx 实际通过 127.0.0.1 连接 Go 服务，可以填写：

合并配置片段config.prod.yaml · 同机 Nginx 示例

  # 保留现有的 port 等配置
  trusted_proxies:
    - "127.0.0.1"

trusted_proxies 可以填写多个 IP 或 CIDR 网段。空列表、未配置或配置为 null 时，传给 Gin 的列表为空，等同于不信任任何代理；同机代理环境必须填写实际代理地址，否则多个用户可能被识别为同一个代理 IP。地址格式有误时，SetTrustedProxies 会返回错误并阻止服务启动。

下面的 main.go 从配置中读取可信代理，并将同一个限流器实例挂到登录路由。它仍是同机代理版本：Go 服务只监听 127.0.0.1。服务与代理不在同一台主机或容器网络命名空间时，需要按实际网络调整监听地址及可信代理。

替换完整文件main.go
package main

import (
	"SolarNotesAdminAPI/conf"
	"SolarNotesAdminAPI/controller/v1/admin"
	"SolarNotesAdminAPI/db"
	"SolarNotesAdminAPI/middleware"
	"SolarNotesAdminAPI/snowflake"
	"SolarNotesAdminAPI/validation"
	"log"

	"github.com/gin-gonic/gin"
)

// main 加载配置与依赖，注册校验和登录路由，并启动仅供代理访问的服务。
func main() {
	err := conf.Load()
	if err != nil {
		log.Fatal(err)
	}
	configInstance := conf.Get()

	err = validation.Register()
	if err != nil {
		log.Fatal(err)
	}

	err = db.Connect(configInstance.Database)
	if err != nil {
		log.Fatal(err)
	}

	err = snowflake.InitSnowflake(configInstance.Snowflake.NodeID)
	if err != nil {
		log.Fatal(err)
	}

	router := gin.Default()
	// 根据配置识别可信代理；空列表表示不信任代理。
	if err := router.SetTrustedProxies(configInstance.Service.TrustedProxies); err != nil {
		log.Fatalf("可信代理配置失败: %v", err)
	}
	router.Use(middleware.Cors())

	v1 := router.Group("/v1")
	loginLimiter := middleware.NewLoginRateLimiter()

	{
		v1.POST("/admin/login", loginLimiter.Handle, admin.Login)
	}

	// 本服务只在回环地址监听；公网 HTTPS 由前置代理提供。
	if err := router.Run("127.0.0.1:" + configInstance.Service.Port); err != nil {
		log.Fatalf("HTTP 服务启动失败: %v", err)
	}
}

没有反向代理、浏览器直接连接 Go 服务的本地开发环境，应将 trusted_proxies 设为空列表，让 Gin 使用连接的对端 IP；不能信任客户端自行发送的转发头。实际有代理时也只填写明确可信的代理地址，并确保代理覆盖而非透传客户端提供的头。信任 127.0.0.1 也意味着信任从本机直接连接 Go 服务的其他进程，应结合实际部署限制访问。

Nginx HTTPS 模板

这是放在 Nginx http 上下文中的服务配置示例，不是工程文件。将域名、证书和端口替换为实际值；证书应由客户端信任，Go 端口须与 service.port 一致。前端必须从一开始就使用 HTTPS 发送登录请求。

部署模板Nginx · HTTPS 代理示例
# 示例：与 Go 服务同机运行的 HTTPS 反向代理。
# 按部署环境替换域名、证书路径和 Go 服务端口。
server {
    listen 443 ssl;
    server_name admin-api.example.com;

    ssl_certificate     /etc/nginx/tls/admin-api.fullchain.pem;
    ssl_certificate_key /etc/nginx/tls/admin-api.key;

    location / {
        proxy_http_version 1.1;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $remote_addr;
        proxy_set_header X-Forwarded-Proto  $scheme;
        proxy_pass http://127.0.0.1:4060;
    }
}

本示例用 $remote_addr 覆盖客户端传来的 X-Forwarded-For 和 X-Real-IP。如果 Nginx 前面还有负载均衡器或 CDN，$remote_addr 可能是上游代理；应先按实际可信网段配置真实客户端 IP，不能直接信任来自公网的转发头。

配置真实证书后，在部署环境执行 nginx -t 再按该环境的流程加载配置；本文没有启动或修改 Nginx，也没有验证实际证书和生产网络。

05 / DIAGNOSTICS

让日志帮助定位错误

DatabaseError.Error() 与 InternalError.Error() 只返回统一消息，直接打印它们看不到原因。下面读取包装的 Cause，将已知原因映射成受控类别；MySQL 错误记录数字错误码，不输出驱动返回的原始消息。

原因日志中的信息
MySQL 错误reason=mysql mysql_code=...
请求取消 / 超时context_canceled / context_deadline_exceeded
网络错误reason=network
密码摘要格式无效reason=invalid_password_hash
其他原因reason=unclassified 和 Go 类型；不包含任意错误文本。

未知错误只记录类型，可能仍不足以定位具体故障；可继续为可信的错误类型增加分类，或通过受控、脱敏的内部诊断渠道排查。不要把请求体、密码、JWT、密钥、连接字符串或未经审查的原始错误直接写入日志。

package diagnostics

import (
	"context"
	"errors"
	"fmt"
	"log"
	"net"

	"SolarNotesAdminAPI/auth"
	"SolarNotesAdminAPI/sys_error"
	"github.com/go-sql-driver/mysql"
)

// LogAdminLoginFailure 根据包装类型记录管理员登录故障的安全诊断信息。
// 不记录原始错误文本、凭据、JWT 或请求体。
func LogAdminLoginFailure(err error) {
	var dbErr *sys_error.DatabaseError
	var internalErr *sys_error.InternalError
	switch {
	case errors.As(err, &dbErr):
		log.Printf("管理员登录失败: category=database %s", describeLoginCause(dbErr.Cause))
	case errors.As(err, &internalErr):
		log.Printf("管理员登录失败: category=internal %s", describeLoginCause(internalErr.Cause))
	default:
		log.Printf("管理员登录失败: category=unknown %s", describeLoginCause(err))
	}
}

// describeLoginCause 将底层原因映射成受控类别，MySQL 错误只包含数字错误码。
// 对未知错误仅保留类型；需要更多细节时应通过受控、脱敏的内部诊断途径排查。
func describeLoginCause(cause error) string {
	switch {
	case cause == nil:
		return "reason=none"
	case errors.Is(cause, context.Canceled):
		return "reason=context_canceled"
	case errors.Is(cause, context.DeadlineExceeded):
		return "reason=context_deadline_exceeded"
	case errors.Is(cause, auth.InvalidHashErr):
		return "reason=invalid_password_hash"
	}

	var mysqlErr *mysql.MySQLError
	if errors.As(cause, &mysqlErr) && mysqlErr != nil {
		return fmt.Sprintf("reason=mysql mysql_code=%d", mysqlErr.Number)
	}
	var networkErr *net.OpError
	if errors.As(cause, &networkErr) && networkErr != nil {
		return "reason=network"
	}
	return fmt.Sprintf("reason=unclassified cause_type=%T", cause)
}

package admin

import (
	"SolarNotesAdminAPI/biz"
	"SolarNotesAdminAPI/diagnostics"
	"SolarNotesAdminAPI/param/req/v1/admin"
	respAdmin "SolarNotesAdminAPI/param/resp/v1/admin"
	"SolarNotesAdminAPI/resp_body"
	"SolarNotesAdminAPI/sys_error"
	"errors"
	"net/http"
	"strconv"

	"github.com/gin-gonic/gin"
)

// Login 本函数用于处理管理员登录行为
func Login(c *gin.Context) {
	c.Header("Cache-Control", "no-store")
	req := &admin.LoginReq{}
	respBody := &resp_body.RespBody{}

	err := c.ShouldBindJSON(req)
	if err != nil {
		respBody.ValidateError(err)
		c.JSON(http.StatusOK, respBody)
		return
	}

	adminBiz := &biz.Admin{}
	err = adminBiz.Login(c.Request.Context(), req)
	if err != nil {
		var dbErr *sys_error.DatabaseError
		switch {
		case errors.Is(err, sys_error.InvalidLoginRequestErr):
			respBody.ValidateError(err)
		case errors.Is(err, sys_error.InvalidCredentialsErr):
			respBody.LoginCredentialError()
		case errors.Is(err, sys_error.LoginStateChangedErr):
			respBody.LoginStateChangedError()
		case errors.As(err, &dbErr):
			diagnostics.LogAdminLoginFailure(err)
			respBody.DBError(err)
		default:
			diagnostics.LogAdminLoginFailure(err)
			respBody.InternalError(err)
		}

		c.JSON(http.StatusOK, respBody)
		return
	}

	loginResp := &respAdmin.LoginResp{}
	loginResp.Admin = &respAdmin.LoginAdminResp{
		ID:          strconv.FormatInt(adminBiz.ID, 10),
		Username:    adminBiz.Username,
		AccessToken: adminBiz.AccessToken,
	}

	data := map[string]interface{}{
		"admin": loginResp.Admin,
	}
	respBody.Success(data)

	c.JSON(http.StatusOK, respBody)
}

// testAllowedLogin 标记请求已经到达登录处理器，不访问真实数据库。
func testAllowedLogin(c *gin.Context) { c.Status(http.StatusNoContent) }

// testConcurrentAttempt 执行一次并发限流判断并记录放行数量。
func testConcurrentAttempt(l *LoginRateLimiter, wg *sync.WaitGroup, accepted *atomic.Int32) {
	defer wg.Done()
	if l.allow("same-client") {
		accepted.Add(1)
	}
}

// TestLoginLimiterWindowAndCleanup 验证次数限制、窗口过期和客户端上限。
func TestLoginLimiterWindowAndCleanup(t *testing.T) {
	l := NewLoginRateLimiter()
	for i := 0; i < loginMaxRequests; i++ {
		if !l.allow("one") {
			t.Fatal("blocked too soon")
		}
	}
	if l.allow("one") {
		t.Fatal("sixth request must be blocked")
	}
	if !l.allow("two") {
		t.Fatal("a different client was blocked")
	}
	l.requests["one"] = []time.Time{time.Now().Add(-loginWindow - time.Second)}
	if !l.allow("one") {
		t.Fatal("expired window was not released")
	}

	full := NewLoginRateLimiter()
	full.lastCleanup = time.Now()
	for i := 0; i < loginMaxClients; i++ {
		full.requests[fmt.Sprint(i)] = []time.Time{time.Now()}
	}
	if full.allow("new-client") || len(full.requests) != loginMaxClients {
		t.Fatal("client limit should reject a new client without growing the map")
	}
	for key := range full.requests {
		full.requests[key] = []time.Time{time.Now().Add(-loginWindow - time.Second)}
	}
	full.lastCleanup = time.Now().Add(-loginWindow - time.Second)
	if !full.allow("new-client") {
		t.Fatal("expired clients were not cleaned")
	}
}

// TestLoginLimiterConcurrent 验证同一客户端并发请求不会突破限制。
func TestLoginLimiterConcurrent(t *testing.T) {
	l := NewLoginRateLimiter()
	var accepted atomic.Int32
	var wg sync.WaitGroup
	for i := 0; i < 100; i++ {
		wg.Add(1)
		go testConcurrentAttempt(l, &wg, &accepted)
	}
	wg.Wait()
	if accepted.Load() != int32(loginMaxRequests) {
		t.Fatalf("accepted %d requests", accepted.Load())
	}
}

// TestLoginLimiterResponseAndUntrustedIP 验证响应约定和不可信转发头不能绕过限制。
func TestLoginLimiterResponseAndUntrustedIP(t *testing.T) {
	gin.SetMode(gin.TestMode)
	router := gin.New()
	if err := router.SetTrustedProxies(nil); err != nil {
		t.Fatal(err)
	}
	l := NewLoginRateLimiter()
	router.POST("/login", l.Handle, testAllowedLogin)
	for i := 0; i <= loginMaxRequests; i++ {
		req := httptest.NewRequest("POST", "/login", nil)
		req.RemoteAddr = "198.51.100.10:4321"
		req.Header.Set("X-Forwarded-For", fmt.Sprintf("192.0.2.%d", i+1))
		out := httptest.NewRecorder()
		router.ServeHTTP(out, req)
		if out.Header().Get("Cache-Control") != "no-store" {
			t.Fatal("missing no-store")
		}
		if i < loginMaxRequests && out.Code != http.StatusNoContent {
			t.Fatalf("status=%d", out.Code)
		}
		if i == loginMaxRequests {
			var body resp_body.RespBody
			if out.Code != http.StatusOK {
				t.Fatalf("status=%d", out.Code)
			}
			if err := json.Unmarshal(out.Body.Bytes(), &body); err != nil {
				t.Fatal(err)
			}
			if body.Code != resp_body.LoginRateLimitCode || body.Data == nil || len(body.Data) != 0 {
				t.Fatalf("unexpected rate-limit response: %+v", body)
			}
		}
	}
}

// TestLoginLimiterTrustedProxy 验证可信代理转发的不同客户端分别计数。
func TestLoginLimiterTrustedProxy(t *testing.T) {
	gin.SetMode(gin.TestMode)
	router := gin.New()
	if err := router.SetTrustedProxies([]string{"127.0.0.1"}); err != nil {
		t.Fatal(err)
	}
	limiter := NewLoginRateLimiter()
	router.POST("/login", limiter.Handle, testAllowedLogin)
	for i := 0; i <= loginMaxRequests; i++ {
		req := httptest.NewRequest("POST", "/login", nil)
		req.RemoteAddr = "127.0.0.1:4321"
		req.Header.Set("X-Forwarded-For", "198.51.100.10")
		out := httptest.NewRecorder()
		router.ServeHTTP(out, req)
		if i == loginMaxRequests && out.Code != http.StatusOK {
			t.Fatal("proxied client should be rate limited")
		}
	}
	req := httptest.NewRequest("POST", "/login", nil)
	req.RemoteAddr = "127.0.0.1:4321"
	req.Header.Set("X-Forwarded-For", "198.51.100.11")
	out := httptest.NewRecorder()
	router.ServeHTTP(out, req)
	if out.Code != http.StatusNoContent {
		t.Fatal("another proxied client should have its own quota")
	}
}

package diagnostics

import (
	"bytes"
	"errors"
	"log"
	"strings"
	"testing"

	"SolarNotesAdminAPI/sys_error"
	"github.com/go-sql-driver/mysql"

// TestLoginErrorDiagnostics 验证错误码可用于排查，敏感错误文本不会进入日志。
func TestLoginErrorDiagnostics(t *testing.T) {
	var output bytes.Buffer
	previous := log.Writer()
	log.SetOutput(&output)
	defer log.SetOutput(previous)

	LogAdminLoginFailure(&sys_error.DatabaseError{
		Cause: &mysql.MySQLError{Number: 1146, Message: "DO_NOT_LOG"},
	})
	if !strings.Contains(output.String(), "mysql_code=1146") || strings.Contains(output.String(), "DO_NOT_LOG") {
		t.Fatal("database log leaked the message or omitted the code")
	}
	output.Reset()
	LogAdminLoginFailure(&sys_error.InternalError{Cause: errors.New("DO_NOT_LOG")})
	if strings.Contains(output.String(), "DO_NOT_LOG") {
		t.Fatal("internal log leaked the error message")
	}
}
