新增完整文件conf/jwt.go
package conf

import (
	"encoding/base64"
	"errors"
	"strings"
	"time"
)

// JWT 保存 YAML 中的令牌配置与启动时解码得到的私有密钥。
type JWT struct {
	// Issuer 对应 JWT 的 iss，标识允许签发令牌的后端。
	Issuer string `yaml:"issuer"`
	// Audience 对应 JWT 的 aud，标识令牌预期服务的客户端。
	Audience string `yaml:"audience"`
	// ExpireSeconds 是令牌有效秒数；必须为正数且能安全转换为 time.Duration。
	ExpireSeconds int64 `yaml:"expire_seconds"`
	// Secret 对应 jwt.secret，保存 Base64 编码的随机密钥；不得输出到日志或响应。
	Secret string `yaml:"secret"`
	// signingKey 保存从 Secret 解码出的原始字节；私有字段不参与 YAML 绑定。
	signingKey []byte
}

// validate 在配置加载阶段检查签发方、受众、有效期和 YAML 中的 Base64 密钥。
// 成功后将至少 32 字节的密钥保存到接收者；失败返回原因并阻止配置加载。
func (j *JWT) validate() error {
	j.Issuer = strings.TrimSpace(j.Issuer)
	j.Audience = strings.TrimSpace(j.Audience)
	if j.Issuer == "" || j.Audience == "" {
		return errors.New("jwt.issuer 和 jwt.audience 不能为空")
	}
	const maxSeconds = int64((1<<63 - 1) / int64(time.Second))
	if j.ExpireSeconds <= 0 || j.ExpireSeconds > maxSeconds {
		return errors.New("jwt.expire_seconds 必须是有效的正整数秒数")
	}

	encoded := strings.TrimSpace(j.Secret)
	if encoded == "" {
		return errors.New("jwt.secret 不能为空")
	}
	key, err := base64.StdEncoding.DecodeString(encoded)
	if err != nil || len(key) < 32 {
		return errors.New("jwt.secret 必须是合法 Base64，解码后至少 32 字节")
	}
	j.signingKey = append([]byte(nil), key...)
	return nil
}

// SigningKey 返回已校验密钥的副本，防止调用方修改配置内部的字节切片。
func (j *JWT) SigningKey() []byte {
	return append([]byte(nil), j.signingKey...)
}

// Lifetime 将已校验的有效秒数转为 time.Duration，供 JWT 签发工具使用。
func (j *JWT) Lifetime() time.Duration {
	return time.Duration(j.ExpireSeconds) * time.Second
}

实现摘要编码与恒定时间比较

这份文件沿用已确认的数据库约定：password_hash 为原始密码的 SHA-256 小写十六进制摘要，token_hash 为完整 JWT 的同类摘要。两者输入不同、用途不同，函数可以复用。

新增完整文件auth/hash.go
package auth

import (
	"crypto/sha256"
	"crypto/subtle"
	"encoding/hex"
	"errors"
)

// ErrInvalidHash 表示数据库摘要无法解码为 32 字节，区别于普通密码不匹配。
var ErrInvalidHash = errors.New("invalid SHA-256 hexadecimal digest")

// SHA256Hex 计算 value 的 SHA-256 摘要，返回 64 个小写十六进制字符。
// 调用方传原始密码时得到密码摘要，传完整 JWT 时得到数据库令牌摘要。
func SHA256Hex(value string) string {
	digest := sha256.Sum256([]byte(value))
	return hex.EncodeToString(digest[:])
}

// MatchSHA256 将 value 的摘要与 expectedHex 解码后的 32 字节摘要进行常量时间比较。
// 匹配返回 true, nil，不匹配返回 false, nil；存储摘要格式损坏时返回 ErrInvalidHash。
func MatchSHA256(value, expectedHex string) (bool, error) {
	expected, err := hex.DecodeString(expectedHex)
	if err != nil || len(expected) != sha256.Size {
		return false, ErrInvalidHash
	}
	actual := sha256.Sum256([]byte(value))
	return subtle.ConstantTimeCompare(actual[:], expected) == 1, nil
}

SHA256Hex 始终生成 64 个字符；MatchSHA256 先检查数据库摘要能否解码为 32 字节，再比较字节。错误摘要编码与密码不匹配分开返回：前者是内部数据异常，后者是正常的凭据错误。

本实现保持现有密码算法，未做密码数据迁移。模型中的密码字段改名为 PasswordHash，使调用代码不会把数据库摘要误当成原始密码。

实现 JWT 签发与基础验证工具

JWTOptions 只包含工具需要的可信参数，工具包不读取数据库。签发时生成随机 jti，管理员主键作为字符串 sub；这样同一秒内两次登录也会产生不同的 JWT。

JWT 工具按职责拆分为三个文件：参数结构体、校验与验签回调方法；声明结构体与校验方法；独立令牌函数与错误常量。三个文件均属于 auth 包，业务层与测试中的现有调用方式保持一致。后续落地时，将独立函数、错误和常量从 admin_claims.go 移至 token.go；若仍保留原来的 auth/jwt.go，应一并移除旧定义，避免重复声明。

auth/jwt_options.go：参数结构体、校验与验签回调

包含 JWTOptions、validate() 与 verificationKey()，集中定义可信签发参数、校验参数并提供验签密钥。回调需要读取当前配置中的密钥，因此作为 JWTOptions 的方法；解析器直接接收 options.verificationKey，无需在调用处定义匿名函数。

新增完整文件auth/jwt_options.go
package auth

import (
	"errors"
	"strings"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

// JWTOptions 定义签发与解析工具所需的可信参数，不读取数据库或请求头。
type JWTOptions struct {
	// Issuer 是期望的 iss；签发时写入，解析时要求完全匹配。
	Issuer string
	// Audience 是期望的 aud；签发时写入，解析时要求包含此受众。
	Audience string
	// Lifetime 是签发有效期，要求至少 1 秒且为整数秒。
	Lifetime time.Duration
	// SigningKey 是至少 32 字节的原始密钥，用于 HS256 签名与验签。
	SigningKey []byte
}

// validate 检查 JWTOptions 是否满足工具的参数约束；失败返回配置错误。
// 签发与解析都先执行此检查，不接受请求方自行指定算法或密钥。
func (o JWTOptions) validate() error {
	if strings.TrimSpace(o.Issuer) == "" || strings.TrimSpace(o.Audience) == "" {
		return errors.New("JWT issuer and audience are required")
	}
	if o.Lifetime < time.Second || o.Lifetime%time.Second != 0 {
		return errors.New("JWT lifetime must be positive whole seconds")
	}
	if len(o.SigningKey) < 32 {
		return errors.New("JWT signing key must contain at least 32 bytes")
	}
	return nil
}

// verificationKey 为 JWT 解析器提供当前配置中的 HS256 验签密钥。
// token 由解析器传入；算法不符合要求时返回 ErrInvalidToken。
// 方法值 options.verificationKey 可直接作为回调，携带该次调用的配置，无需全局变量。
func (o JWTOptions) verificationKey(token *jwt.Token) (any, error) {
	if token.Method != jwt.SigningMethodHS256 {
		return nil, ErrInvalidToken
	}
	return o.SigningKey, nil
}

package auth

import (
	"strconv"

	"github.com/golang-jwt/jwt/v5"
)

// AdminClaims 在标准 JWT 声明之外增加管理员身份的自定义校验。
type AdminClaims struct {
	// RegisteredClaims 承载 iss、sub、aud、iat、exp、jti 等标准字段。
	// 其中 sub 是管理员 ID 的十进制字符串，jti 是每次签发生成的随机标识。
	jwt.RegisteredClaims
}

// Validate 在 JWT 库完成标准声明检查后，补充管理员身份约束。
// sub 必须能解析为正数 int64，jti 与 iat 必须存在；失败返回 ErrInvalidToken。
func (c AdminClaims) Validate() error {
	id, err := strconv.ParseInt(c.Subject, 10, 64)
	if err != nil || id <= 0 || c.ID == "" || c.IssuedAt == nil {
		return ErrInvalidToken
	}
	return nil
}

package auth

import (
	"crypto/rand"
	"encoding/hex"
	"errors"
	"strconv"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

var (
	// ErrInvalidToken 表示令牌格式、签名或声明不符合要求。
	ErrInvalidToken = errors.New("invalid administrator token")
	// ErrExpiredToken 表示验签与其他检查正常，唯一的具体问题是过期。
	ErrExpiredToken = errors.New("administrator token expired")
)

// maxAdminTokenLength 是本实现对完整 JWT 的字节长度限制，并非 JWT 规范的固定长度。
const maxAdminTokenLength = 4096

// IssueAdminToken 为正数 adminID 签发完整的 HS256 JWT。
// options 提供签发方、受众、有效期和密钥；每次生成随机 jti，避免同秒登录得到相同令牌。
// 成功返回完整 JWT；参数、随机数、签名或长度检查失败时返回错误，不写数据库。
func IssueAdminToken(adminID int64, options JWTOptions) (string, error) {
	if err := options.validate(); err != nil {
		return "", err
	}
	if adminID <= 0 {
		return "", errors.New("administrator ID must be positive")
	}

	randomID := make([]byte, 16)
	if _, err := rand.Read(randomID); err != nil {
		return "", err
	}
	now := time.Now().UTC().Truncate(time.Second)
	claims := AdminClaims{RegisteredClaims: jwt.RegisteredClaims{
		Issuer:    options.Issuer,                                // iss：当前后端签发方。
		Subject:   strconv.FormatInt(adminID, 10),                // sub：用字符串存储管理员主键，避免精度损失。
		Audience:  jwt.ClaimStrings{options.Audience},            // aud：这枚令牌允许面向的客户端。
		IssuedAt:  jwt.NewNumericDate(now),                       // iat：本次签发时间，统一使用 UTC。
		ExpiresAt: jwt.NewNumericDate(now.Add(options.Lifetime)), // exp：后端计算的到期时间，解析时强制要求并校验。
		ID:        hex.EncodeToString(randomID),                  // jti：本次签发的随机唯一标识。
	}}
	token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
	raw, err := token.SignedString(options.SigningKey)
	if err != nil {
		return "", err
	}
	if len(raw) > maxAdminTokenLength {
		return "", errors.New("JWT exceeds the supported length")
	}
	return raw, nil
}

// ParseAdminToken 验证 raw 中完整 JWT 的签名、算法、标准声明与管理员身份。
// options 提供可信配置；成功返回管理员 ID，单独过期返回 ErrExpiredToken，
// 其他令牌错误返回 ErrInvalidToken，配置错误原样返回。
// 本方法不查询数据库；调用方仍须比较当前 token_hash，才能判断令牌是否已撤销。
func ParseAdminToken(raw string, options JWTOptions) (int64, error) {
	if err := options.validate(); err != nil {
		return 0, err
	}
	if raw == "" || len(raw) > maxAdminTokenLength {
		return 0, ErrInvalidToken
	}

	claims := &AdminClaims{}
	// 直接传入具名方法值，方法从当前 options 取得密钥；签名与声明由库继续校验。
	token, err := jwt.ParseWithClaims(raw, claims, options.verificationKey,
		jwt.WithValidMethods([]string{jwt.SigningMethodHS256.Alg()}),
		jwt.WithIssuer(options.Issuer),
		jwt.WithAudience(options.Audience),
		jwt.WithExpirationRequired(),
		jwt.WithIssuedAt(),
	)
	if err != nil {
		if isOnlyExpired(err) {
			return 0, ErrExpiredToken
		}
		return 0, ErrInvalidToken
	}
	if token == nil || !token.Valid {
		return 0, ErrInvalidToken
	}
	id, err := strconv.ParseInt(claims.Subject, 10, 64)
	if err != nil || id <= 0 {
		return 0, ErrInvalidToken
	}
	return id, nil
}

// isOnlyExpired 判断 JWT 库包装或合并的错误是否仅包含过期这一项具体错误。
// 过期还伴随错误签发方、受众或主体时返回 false，避免误报为单纯过期。
func isOnlyExpired(err error) bool {
	hasExpired, hasOther := classifyTokenError(err)
	return hasExpired && !hasOther
}

// classifyTokenError 递归展开错误链，分别报告是否出现过期错误和其他具体错误。
// hasExpired 表示存在 ErrTokenExpired，hasOther 表示存在其他具体错误。
// 多个分支的结果按逻辑或合并；nil 和 JWT 库的通用声明错误标记不计为具体错误。
// 函数仅通过参数与返回值传递状态，不使用闭包或包级可变状态。
func classifyTokenError(err error) (hasExpired, hasOther bool) {
	if err == nil {
		return false, false
	}
	switch wrapped := err.(type) {
	case interface{ Unwrap() []error }:
		for _, child := range wrapped.Unwrap() {
			childExpired, childOther := classifyTokenError(child)
			hasExpired = hasExpired || childExpired
			hasOther = hasOther || childOther
		}
		return hasExpired, hasOther
	case interface{ Unwrap() error }:
		return classifyTokenError(wrapped.Unwrap())
	default:
		// 通过 errors.Is 匹配具体错误，外层递归仍负责汇总所有分支。
		switch {
		case errors.Is(err, jwt.ErrTokenExpired):
			return true, false
		case errors.Is(err, jwt.ErrTokenInvalidClaims):
			// JWT 库用于包装具体声明错误的分类标记。
			return false, false
		default:
			return false, true
		}
	}
}
函数 / 方法所在文件作用
JWTOptions.validatejwt_options.go统一校验签发方、受众、有效期和密钥长度。
JWTOptions.verificationKeyjwt_options.go供 ParseWithClaims 直接调用的具名回调，限制算法并返回当前配置中的验签密钥。
AdminClaims.Validateadmin_claims.go在 JWT 库标准校验外要求合法正数管理员 ID、jti 和签发时间。
IssueAdminTokentoken.go签发 HS256 JWT，包含 sub、iss、aud、iat、exp、jti；不向前端返回额外的时间字段。
ParseAdminTokentoken.go验签、限制算法，要求并校验声明；验证通过后才返回 int64 管理员 ID。
isOnlyExpiredtoken.go根据错误分类结果判断是否仅过期；过期同时伴随其他具体错误时返回 false。
classifyTokenErrortoken.go独立递归遍历错误链，通过两个返回值汇总过期与其他错误，不共享可变状态。

ParseAdminToken() 与 isOnlyExpired() 均直接调用外部声明的具名函数或方法；递归函数 classifyTokenError() 通过返回值传递每个分支的分类结果。调用接口及错误码保持一致。

具体错误通过 errors.Is() 匹配，同时保留外层对 Unwrap() error 和 Unwrap() []error 的递归展开。逐个汇总子错误，才能区分“仅过期”和“过期同时伴随其他错误”；只检查是否包含过期错误还不够。

4096 字节是本实现的令牌接收上限，并非 JWT 规范规定的固定长度。签发和解析采用相同限制，配置导致令牌过长时不会返回成功。

ParseAdminToken 只验证 JWT 本身。它不查询当前 token_hash，因此不能单独作为完整鉴权中间件使用。后续路由还必须执行第 11 节的数据库摘要检查。

// 查询成功后将结果写入接收者 a，返回 nil；查询失败时不覆盖接收者。
// 不存在时返回 ErrAdminNotFound，其他数据库错误原样返回；调用方应先处理错误再读取字段。
func (a *Admin) FindAdminByName(ctx context.Context, username string) error {
	if a == nil {
		return errors.New("administrator model is nil")
	}
	if db.DB == nil {
		return errors.New("database is not initialized")
	}
	// 使用临时记录接收查询结果，避免接收者已有 ID 隐式参与查询条件。
	record := &Admin{}
	err := db.DB.WithContext(ctx).
		Select("id", "name", "password_hash").
		Where("name = ?", username).
		Take(record).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		return ErrAdminNotFound
	}
	if err != nil {
		return err
	}
	*a = *record
	return nil
}

// UpdateAdminTokenHash 使用 ctx 为接收者代表的管理员条件更新当前令牌摘要。
// a.ID 和 a.PasswordHash 必须来自已成功完成的查询；PasswordHash 保留查询时的值，
// 用作密码未发生变化的更新条件，防止改密后旧密码验证结果重新建立登录。
// newTokenHash 是新 JWT 的摘要；命中一行时同步 a.TokenHash 并返回 true，
// 未命中返回 false, nil，执行失败返回错误；失败时不修改接收者字段。
func (a *Admin) UpdateAdminTokenHash(ctx context.Context, newTokenHash string) (bool, error) {
	if a == nil {
		return false, errors.New("administrator model is nil")
	}
	if db.DB == nil {
		return false, errors.New("database is not initialized")
	}
	result := db.DB.WithContext(ctx).
		Model(&Admin{}).
		Where("id = ?", a.ID).
		Where("CAST(password_hash AS BINARY) = CAST(? AS BINARY)", a.PasswordHash).
		Update("token_hash", newTokenHash)
	if result.Error != nil {
		return false, result.Error
	}
	if result.RowsAffected != 1 {
		return false, nil
	}
	a.TokenHash = newTokenHash
	return true, nil
}

Model 把 GORM 的“记录不存在”转换为 ErrAdminNotFound，让 Biz 不必依赖 ORM 的错误类型。账号查询使用参数占位符，区分大小写的比较由现有列排序规则执行。

两个数据库操作都使用 *model.Admin 接收者。Biz 每次创建自己的模型实例，先调用 record.FindAdminByName(ctx, username) 并处理错误，再读取该实例中的身份与密码摘要；随后通过同一个实例调用 record.UpdateAdminTokenHash(ctx, newTokenHash)。

模型方法返回值与接收者变化
FindAdminByName(ctx, username)返回 error；成功时将查询结果写入当前实例，失败时不覆盖原字段。
UpdateAdminTokenHash(ctx, newTokenHash)返回 (bool, error)；使用实例 ID 与查询时的密码摘要作为更新条件，成功后同步实例 TokenHash。

为什么更新时还要带密码哈希条件

查询账号后，管理员可能正好改密。模型实例的 PasswordHash 保留本次查询时的摘要，更新前不要修改该字段。写入令牌时再次要求数据库密码摘要与此值一致，可避免旧密码验证结果在改密提交后重新建立登录。CAST 条件按编码内容精确比较，更新仍通过实例的主键定位。

UPDATE admin
SET token_hash = ?
WHERE id = ?
  AND CAST(password_hash AS BINARY) = CAST(? AS BINARY);

更新只命中一行才算保存成功；0 行由 Biz 转成“账号状态已变化”。使用单列 Update，不把查询出的旧模型整体 Save 回去。模型未声明日期字段，现有数据库的 ON UPDATE 继续维护更新时间。

package validation

import (
	"errors"
	"reflect"
	"strings"

	"github.com/gin-gonic/gin/binding"
	"github.com/go-playground/validator/v10"
)

// Register 为 Gin 的默认参数校验器注册自定义规则。
// 应在程序启动时、任何请求参数校验开始前调用一次；注册失败时返回错误。
func Register() error {
	if binding.Validator == nil {
		return errors.New("Gin 参数校验器未启用")
	}

	v, ok := binding.Validator.Engine().(*validator.Validate)
	if !ok || v == nil {
		return errors.New("无法获取 Gin 默认参数校验器")
	}

	return v.RegisterValidation("notblank", notBlank)
}

// notBlank 检查字符串是否至少包含一个非空白字符；非字符串字段不通过校验。
// 它只读取字段，使用 TrimSpace 判断，不修改原始用户名或密码。
func notBlank(field validator.FieldLevel) bool {
	value := field.Field()
	if value.Kind() != reflect.String {
		return false
	}
	return strings.TrimSpace(value.String()) != ""
}

package biz

import (
	"context"
	"errors"

	"SolarNotesAdminAPI/auth"
	"SolarNotesAdminAPI/conf"
	"SolarNotesAdminAPI/model"
	reqAdmin "SolarNotesAdminAPI/param/req/v1/admin"
	"SolarNotesAdminAPI/sys_error"
)

// Admin 是每次登录请求独立创建的业务实体，登录成功后保存已确认的身份和本次令牌。
type Admin struct {
	// ID 保存已验证管理员的主键，业务层使用 int64。
	ID int64
	// Username 保存数据库返回的账号名称，不保存请求中的原始密码。
	Username string
	// AccessToken 是本次签发的完整 JWT，仅在其摘要成功写入数据库后设置。
	AccessToken string
}

// Login 使用 ctx 执行管理员登录，req 接收 Controller 绑定后的完整请求对象。
// 查询账号、验证密码、签发 JWT 并条件保存摘要；成功后将结果写入当前 Admin 并返回 nil。
// 失败仅返回 sys_error 中的错误，Controller 负责处理响应；实体不保存原始密码。
func (a *Admin) Login(ctx context.Context, req *reqAdmin.LoginReq) error {
	if a == nil {
		return &sys_error.InternalError{Cause: errors.New("nil business entity")}
	}
	// 如果实例被重复使用，先清除旧结果，避免失败后保留上一枚令牌。
	a.ID, a.Username, a.AccessToken = 0, "", ""

	// Controller 已执行字段校验；这里防护直接调用时的 nil。
	if ctx == nil || req == nil || req.Admin == nil {
		return sys_error.ErrInvalidLoginRequest
	}

	record := &model.Admin{}
	err := record.FindAdminByName(ctx, req.Admin.Username)
	if errors.Is(err, model.ErrAdminNotFound) {
		return sys_error.ErrInvalidCredentials
	}
	if err != nil {
		return &sys_error.DatabaseError{Cause: err}
	}

	matched, err := auth.MatchSHA256(req.Admin.Password, record.PasswordHash)
	if err != nil {
		return &sys_error.InternalError{Cause: err}
	}
	if !matched {
		return sys_error.ErrInvalidCredentials
	}

	config := conf.Get()
	if config == nil {
		return &sys_error.InternalError{Cause: errors.New("configuration is not loaded")}
	}
	options := auth.JWTOptions{
		Issuer:     config.JWT.Issuer,
		Audience:   config.JWT.Audience,
		Lifetime:   config.JWT.Lifetime(),
		SigningKey: config.JWT.SigningKey(),
	}
	token, err := auth.IssueAdminToken(record.ID, options)
	if err != nil {
		return &sys_error.InternalError{Cause: err}
	}

	updated, err := record.UpdateAdminTokenHash(ctx, auth.SHA256Hex(token))
	if err != nil {
		return &sys_error.DatabaseError{Cause: err}
	}
	if !updated {
		return sys_error.ErrLoginStateChanged
	}

	a.ID, a.Username, a.AccessToken = record.ID, record.Username, token
	return nil
}

成功顺序保持固定

  1. 创建 model.Admin 实例，调用 FindAdminByName 查询账号；不存在与密码错误均返回统一凭据错误。
  2. 验证实例中的密码摘要；损坏的数据库摘要按内部错误处理。
  3. 从已加载配置构造 JWTOptions，生成完整 JWT。
  4. 计算 JWT 摘要，调用同一模型实例的 UpdateAdminTokenHash 条件更新管理员记录。
  5. 仅在写库成功后，将 ID、用户名和令牌写入当前 biz.Admin，并返回 nil。

sys_error.DatabaseError 与 sys_error.InternalError 保留内部原因供受控排查，Error() 只返回统一消息。Controller 通过 errors.Is 识别登录业务错误，通过 errors.As 识别数据库错误，并负责选择响应。

const (

	SuccessCode = 200
	// ServiceInternalErrorCode 表示内部处理失败。
	ServiceInternalErrorCode = 500
	// ValidateErrorCode 表示 JSON 绑定或参数校验失败。
	ValidateErrorCode = 10101
	// DBErrorCode 表示数据库操作失败。
	DBErrorCode = 10201
	// LoginCredentialErrorCode 统一表示账号或密码错误。
	LoginCredentialErrorCode = 20001
	// LoginStateChangedCode 表示登录期间凭据状态变化，需要重新登录。
	LoginStateChangedCode = 20002
	// TokenExpiredCode 用于后续受保护请求的令牌过期响应。
	TokenExpiredCode = 20003
	// InvalidTokenCode 用于后续受保护请求的无效或撤销令牌响应。
	InvalidTokenCode = 20004
)

// RespBody 是统一 JSON 响应包装器；本工程使用 HTTP 200 搭配业务 Code。
type RespBody struct {
	// Code 是业务结果码；200 表示成功，其余按约定的错误类别解释。
	Code int `json:"code"`
	// Message 是前端展示或诊断用的说明；成功固定为 success。
	Message string `json:"message"`
	// Data 是业务对象容器；约定直接子项均为对象，错误时赋值空 map 输出 {}。
	Data map[string]any `json:"data"`
}

// Success 设置成功业务码与消息，并接收对象容器 data。
// 调用方需保证 map 的直接值为业务对象；该类型本身不会限制 any 的具体值。
func (r *RespBody) Success(data map[string]any) {
	r.Code = SuccessCode
	r.Message = "success"
	r.Data = data
}

// failure 统一设置错误业务码、消息和空对象 data，避免输出 null 或遗留成功数据。
func (r *RespBody) failure(code int, message string) {
	r.Code, r.Message = code, message
	r.Data = map[string]any{}
}

// ValidateError 生成参数校验错误；err 非空时使用校验消息，否则采用默认说明。
func (r *RespBody) ValidateError(err error) {
	message := "请求参数不合法"
	if err != nil {
		message = err.Error()
	}
	r.failure(ValidateErrorCode, message)
}

// DBError 生成统一数据库错误响应；保留 error 形参以兼容调用方式，不直接暴露其内容。
func (r *RespBody) DBError(_ error) {
	r.failure(DBErrorCode, "数据库操作失败，请稍后重试")
}

// InternalError 生成统一内部错误响应，不向前端输出底层错误。
func (r *RespBody) InternalError(_ error) {
	r.failure(ServiceInternalErrorCode, "服务暂时不可用，请稍后重试")
}

// LoginCredentialError 将账号不存在与密码不匹配统一为相同响应。
func (r *RespBody) LoginCredentialError() {
	r.failure(LoginCredentialErrorCode, "账号或密码错误")
}

// LoginStateChanged 表示验证凭据后，管理员记录的条件更新未命中。
func (r *RespBody) LoginStateChanged() {
	r.failure(LoginStateChangedCode, "账号状态已变化，请重新登录")
}

// TokenExpired 生成过期响应，供后续鉴权中间件在确认令牌仅过期时调用。
func (r *RespBody) TokenExpired() {
	r.failure(TokenExpiredCode, "登录已过期，请重新登录")
}

// InvalidToken 生成令牌无效或已撤销的响应，供后续鉴权中间件使用。
func (r *RespBody) InvalidToken() {
	r.failure(InvalidTokenCode, "登录状态无效，请重新登录")
}

// Login 处理 POST /v1/admin/login；c 提供请求上下文、JSON 绑定与响应写入。
// 绑定完整 LoginReq 后交给 Biz，按错误类别响应；成功时从业务实体构造响应 DTO。
// 无返回值，处理结果通过 HTTP 200 与业务码写入响应，同时禁止缓存登录响应。
func Login(c *gin.Context) {
	c.Header("Cache-Control", "no-store")
	req := &reqAdmin.LoginReq{}
	respBody := &resp_body.RespBody{}

	if err := c.ShouldBindJSON(req); err != nil {
		respBody.ValidateError(err)
		c.JSON(http.StatusOK, respBody)
		return
	}

	adminBiz := &biz.Admin{}
	err := adminBiz.Login(c.Request.Context(), req)
	if err != nil {
		var databaseError *sys_error.DatabaseError
		switch {
		case errors.Is(err, sys_error.ErrInvalidLoginRequest):
			respBody.ValidateError(err)
		case errors.Is(err, sys_error.ErrInvalidCredentials):
			respBody.LoginCredentialError()
		case errors.Is(err, sys_error.ErrLoginStateChanged):
			respBody.LoginStateChanged()
		case errors.As(err, &databaseError):
			log.Printf("管理员登录：%v", databaseError)
			respBody.DBError(databaseError)
		default:
			log.Printf("管理员登录：%v", err)
			respBody.InternalError(err)
		}
		c.JSON(http.StatusOK, respBody)
		return
	}

	loginResp := &respAdmin.LoginResp{
		Admin: &respAdmin.LoginAdminResp{
			ID:          strconv.FormatInt(adminBiz.ID, 10),
			Username:    adminBiz.Username,
			AccessToken: adminBiz.AccessToken,
		},
	}
	respBody.Success(map[string]any{"admin": loginResp.Admin})
	c.JSON(http.StatusOK, respBody)
}

// DB 保存启动时建立的共享连接，Model 通过它发起带上下文的查询。
var DB *gorm.DB

// Connect 根据 dbConf 建立 MySQL/GORM 连接，配置参数化 SQL 日志、单数表名和连接池。
// 成功后将连接保存到 DB；初始化失败时返回错误，交由启动流程决定是否退出。
func Connect(dbConf conf.Database) error {
	dsn := buildDSN(dbConf)
	queryLogger := logger.New(log.New(os.Stdout, "[gorm] ", log.LstdFlags), logger.Config{
		SlowThreshold:             time.Second,
		LogLevel:                  logger.Warn,
		IgnoreRecordNotFoundError: true,
		ParameterizedQueries:      true,
		Colorful:                  false,
	})
	opts := &gorm.Config{
		NamingStrategy: schema.NamingStrategy{SingularTable: true},
		Logger:         queryLogger,
	}

	connection, err := gorm.Open(mysql.Open(dsn), opts)
	if err != nil {
		return fmt.Errorf("连接数据库失败: %w", err)
	}
	sqlDB, err := connection.DB()
	if err != nil {
		return fmt.Errorf("连接数据库失败: %w", err)
	}
	sqlDB.SetMaxOpenConns(10)
	sqlDB.SetMaxIdleConns(5)
	sqlDB.SetConnMaxLifetime(time.Hour)

	DB = connection
	return nil
}

// buildDSN 根据已有数据库配置组装 MySQL 连接字符串，保留字符集、时间解析和时区设置。
// 返回值包含连接凭据，仅交给驱动使用，不应写入日志或响应。
func buildDSN(dbConf conf.Database) string {
	return dbConf.User + ":" + dbConf.Password + "@tcp(" + dbConf.Host +
		":" + dbConf.Port + ")/" + dbConf.Name + "?charset=utf8mb4&parseTime=True&loc=Local"
}

成功后在本地用 auth.SHA256Hex 计算返回令牌的摘要，再用摘要替换下面 SQL 的占位符。查询只返回长度与比较结果，不需要把完整 JWT 作为 SQL 参数发送给数据库。

只读 SQL只读核验 SQL · 用令牌的 SHA-256 摘要替换占位符
SELECT
    CAST(id AS CHAR) AS id,
    name,
    CHAR_LENGTH(token_hash) AS token_hash_length,
    token_hash = '<expected_sha256_hex>' AS token_hash_matches
FROM admin
WHERE name = 'admin';

仍需真实数据库联调确认的行为

  • 正确密码登录后，返回 JWT 的摘要与数据库一致；密码错误不改变现有令牌。
  • 用户名大小写严格依赖当前 name 列规则，Admin 不匹配仅存在的 admin。
  • 密码验证与写摘要之间发生改密时，条件更新未命中，返回 20002。
  • 重复登录覆盖旧摘要；真实路由接入鉴权后，旧令牌、清空摘要及过期令牌均被拒绝。
