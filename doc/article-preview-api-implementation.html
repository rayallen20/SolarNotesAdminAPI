POST /v1/article/preview

点击文件，预览 Markdown

管理员从目录树或右侧文件列表点开文章。管理 API 读取 Markdown，在返回的正文里改写本地图片地址。浏览器渲染正文，并向资源服务器加载图片。

目录管理接口设计 · 第 ⑥ 项 · 目录树返回的文件节点

一次点击如何得到预览

前端从 type === "file" 的节点取 id，以十进制字符串放入请求。发送 Content-Type: application/json 和 Authorization: Bearer <access_token>。目录名、文件路径和资源域名都由服务端确定。

请求中的 catalogue.id 标识点击的文件节点。响应中的 data.article.id 取关联文章的主键 article.id，按十进制字符串返回，以便浏览器精确读取雪花 ID。上例的文章 ID 为示例值。

data.article.title 取目录节点当前的 catalogue.name，保留其原值。content 是 UTF-8 Markdown 字符串，示例里的 \n 是 JSON 对换行的表示。管理员可以预览隐藏书籍、隐藏节点以及尚未排序的文章。

200读取并转换成功。空的 UTF-8 文件也可返回空字符串。
20003 / 20004令牌过期 / 缺失、无效或已撤销。沿用全局管理员鉴权。

按文件节点查询关联文章

article 表没有标题列。article.catalogue_id 是唯一键，关联 catalogue.id，article.path 指向正文。由 (*model.Article).LoadForPreview 以目录表为起点做一次左连接。如果使用内连接，节点已删除和文章记录缺失都会查不到行，控制器就无法分别返回 20301 与 20401。

节点类型使用现有 enum.TypeFile。查询只约束节点 ID，管理员预览不受书籍或节点的显隐、排序影响。工程已启用 GORM 的单数表名策略。查询结束即释放数据库连接，再读取磁盘文件。

新增完整文件model/article.go 
package model

import (
	"context"
	"errors"

	"SolarNotesAdminAPI/db"
	"SolarNotesAdminAPI/enum"

	"gorm.io/gorm"
)

// Article 映射 article 表中预览接口使用的列.
type Article struct {
	ID          int64  `gorm:"column:id;primaryKey;autoIncrement:false"` // ID 是文章主键.
	CatalogueID int64  `gorm:"column:catalogue_id"`                      // CatalogueID 是唯一的文件节点引用.
	Path        string `gorm:"column:path"`                              // Path 是 Markdown 的存储路径.
}

// ArticlePreviewUnavailableErr 表示节点类型或文章记录不满足预览条件.
var ArticlePreviewUnavailableErr = errors.New("article preview unavailable")

// LoadForPreview 按节点 ID 加载文章并返回节点名称. 数据库只读一次.
func (a *Article) LoadForPreview(ctx context.Context, catalogueID int64) (string, error) {
	if db.DB == nil {
		return "", errors.New("database is not initialized")
	}

	// 左连接的文章字段可空, 即使 article 表的对应列定义为 NOT NULL.
	var row struct {
		Name      string  // Name 是 catalogue.name.
		NodeType  string  // NodeType 是 catalogue.type.
		ArticleID *int64  // ArticleID 为 nil 表示没有关联文章.
		Path      *string // Path 只用于后端定位源文件.
	}
	err := db.DB.WithContext(ctx).Model(&Catalogue{}).
		Select("catalogue.name, catalogue.type AS node_type, article.id AS article_id, article.path").
		Joins("LEFT JOIN article ON article.catalogue_id = catalogue.id").
		Where("catalogue.id = ?", catalogueID).
		Take(&row).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		return "", CatalogueNotFoundErr
	}
	if err != nil {
		return "", err
	}
	if row.NodeType != enum.TypeFile || row.ArticleID == nil || row.Path == nil || *row.Path == "" {
		return "", ArticlePreviewUnavailableErr
	}
	*a = Article{ID: *row.ArticleID, CatalogueID: catalogueID, Path: *row.Path}
	return row.Name, nil
}

由 biz.Article 完成一次预览

biz/article.go 包含 Article 结构体及其全部方法。下面是一份可整体复制的文件，按调用顺序列出业务入口、正文读取和图片地址解析。

(*biz.Article).Preview(ctx, req) error 接收完整请求，使用 conf.Get().Storage.RootDir 和 conf.Get().Asset.BaseURL。控制器为每个请求创建一个 biz.Article。Preview 查到文章后调用 a.readPreviewContent，成功后一起填入模型已查出的 orm.ID、预览标题和正文。

存储根句柄、公开资源基址和当前正文目录是本次预览的私有状态。readPreviewContent 通过 defer a.closeAssets() 在读取成功或失败后释放句柄并清空这些字段。同一个实例可顺序预览不同文章。不可预览的内容使用现有 sys_error.ArticlePreviewUnavailableErr 返回 20401；其他磁盘错误包装为 sys_error.InternalError。

package biz

import (
	"context"
	"errors"
	"fmt"
	"io"
	"io/fs"
	"net/url"
	"os"
	"path/filepath"
	"strings"
	"syscall"
	"unicode/utf8"

	"SolarNotesAdminAPI/article_preview"
	"SolarNotesAdminAPI/conf"
	"SolarNotesAdminAPI/model"
	reqArticle "SolarNotesAdminAPI/param/req/v1/article"
	"SolarNotesAdminAPI/sys_error"
)

// Article 保存当前请求已成功读取的文章预览.
type Article struct {
	ID      int64  // ID 是 article 表的主键.
	Title   string // Title 使用文件节点当前显示的名称.
	Content string // Content 保留 Markdown 格式, 供控制器组装响应.

	root       *os.Root // root 限定当前预览读取文件的范围.
	assetBase  *url.URL // assetBase 是从配置解析出的公开资源基址.
	articleDir string   // articleDir 是当前正文所在目录相对于存储根的真实路径.
}

// Preview 从完整请求加载文章, 成功后才更新 ID, Title 和 Content.
func (a *Article) Preview(ctx context.Context, req *reqArticle.PreviewReq) error {
	if a == nil {
		return &sys_error.InternalError{Cause: errors.New("biz.Article.Preview: nil receiver")}
	}
	if ctx == nil {
		return &sys_error.InternalError{Cause: errors.New("biz.Article.Preview: nil context")}
	}
	if req == nil {
		return &sys_error.InternalError{Cause: errors.New("biz.Article.Preview: nil request")}
	}
	if req.Catalogue == nil {
		return &sys_error.InternalError{Cause: errors.New("biz.Article.Preview: nil catalogue reference")}
	}

	cfg := conf.Get()
	if cfg == nil {
		return &sys_error.InternalError{Cause: errors.New("application configuration is not loaded")}
	}
	orm := &model.Article{}
	title, err := orm.LoadForPreview(ctx, req.Catalogue.ID)
	if err != nil {
		switch {
		case errors.Is(err, model.CatalogueNotFoundErr):
			return sys_error.CatalogueNotFoundErr
		case errors.Is(err, model.ArticlePreviewUnavailableErr):
			return sys_error.ArticlePreviewUnavailableErr
		default:
			return &sys_error.DatabaseError{Cause: err}
		}
	}
	content, err := a.readPreviewContent(cfg.Storage.RootDir, cfg.Asset.BaseURL, orm.Path)
	if err != nil {
		if errors.Is(err, sys_error.ArticlePreviewUnavailableErr) {
			return sys_error.ArticlePreviewUnavailableErr
		}
		return &sys_error.InternalError{Cause: err}
	}
	a.ID = orm.ID
	a.Title = title
	a.Content = content
	return nil
}

// readPreviewContent 读取正文, 并把当前文章的图片地址解析方法交给 Markdown 解析器.
func (a *Article) readPreviewContent(rootDir, baseURL, storedPath string) (string, error) {
	err := a.openAssets(rootDir, baseURL)
	if err != nil {
		return "", err
	}
	defer a.closeAssets()

	source, err := a.readMarkdown(rootDir, storedPath)
	if err != nil {
		return "", err
	}
	return article_preview.RewriteImages(source, a.imageURL)
}

// openAssets 在解析文章路径之前打开存储根, 并校验用于拼接的 HTTP(S) 基址.
func (a *Article) openAssets(rootDir, baseURL string) error {
	if !filepath.IsAbs(rootDir) {
		return errors.New("storage.root_dir must be absolute")
	}
	base, err := url.Parse(baseURL)
	if err != nil {
		return err
	}
	if (base.Scheme != "http" && base.Scheme != "https") || base.Host == "" || base.User != nil ||
		base.RawQuery != "" || base.ForceQuery || base.Fragment != "" {
		return errors.New("invalid asset.base_url")
	}
	realRoot, err := filepath.EvalSymlinks(rootDir)
	if err != nil {
		return err
	}
	root, err := os.OpenRoot(realRoot)
	if err != nil {
		return err
	}
	a.root = root
	a.assetBase = base
	return nil
}

// closeAssets 释放存储根并清理本次预览使用的路径信息.
func (a *Article) closeAssets() {
	_ = a.root.Close()
	a.root = nil
	a.assetBase = nil
	a.articleDir = ""
}

// readMarkdown 读取 UTF-8 正文, 成功时记下文章所在目录相对于存储根的真实路径.
func (a *Article) readMarkdown(rootDir, storedPath string) ([]byte, error) {
	name := storedPath
	if filepath.IsAbs(name) {
		relative, err := filepath.Rel(rootDir, name)
		if err != nil {
			return nil, sys_error.ArticlePreviewUnavailableErr
		}
		name = relative
	}
	if !strings.EqualFold(filepath.Ext(name), ".md") {
		return nil, sys_error.ArticlePreviewUnavailableErr
	}
	actual, err := a.resolveStoragePath(name, false)
	if err != nil {
		return nil, err
	}
	if !strings.EqualFold(filepath.Ext(actual), ".md") {
		return nil, sys_error.ArticlePreviewUnavailableErr
	}
	// Linux 和 macOS 上用 O_NONBLOCK 打开, 避免在 Stat 识别 FIFO 之前阻塞.
	f, err := a.root.OpenFile(actual, os.O_RDONLY|syscall.O_NONBLOCK, 0)
	if errors.Is(err, fs.ErrNotExist) {
		return nil, sys_error.ArticlePreviewUnavailableErr
	}
	if err != nil {
		return nil, err
	}
	defer f.Close()
	info, err := f.Stat()
	if err != nil {
		return nil, err
	}
	if !info.Mode().IsRegular() {
		return nil, sys_error.ArticlePreviewUnavailableErr
	}
	content, err := io.ReadAll(f)
	if err != nil {
		return nil, err
	}
	if !utf8.Valid(content) {
		return nil, sys_error.ArticlePreviewUnavailableErr
	}
	a.articleDir = filepath.Dir(actual)
	return content, nil
}

// resolveStoragePath 解析符号链接并检查根边界. 普通缺图可以保留尚不存在的尾部路径.
func (a *Article) resolveStoragePath(name string, allowMissing bool) (string, error) {
	if !filepath.IsLocal(name) || strings.ContainsRune(name, 0) {
		return "", sys_error.ArticlePreviewUnavailableErr
	}
	rootDir := a.root.Name()
	candidate := filepath.Join(rootDir, name)
	var missing []string
	for {
		resolved, err := filepath.EvalSymlinks(candidate)
		if err == nil {
			relative, relErr := filepath.Rel(rootDir, resolved)
			if relErr != nil || !filepath.IsLocal(relative) {
				return "", sys_error.ArticlePreviewUnavailableErr
			}
			for i := len(missing) - 1; i >= 0; i-- {
				relative = filepath.Join(relative, missing[i])
			}
			return relative, nil
		}
		if errors.Is(err, syscall.ENOTDIR) || errors.Is(err, syscall.ELOOP) {
			return "", sys_error.ArticlePreviewUnavailableErr
		}
		if !errors.Is(err, fs.ErrNotExist) || candidate == rootDir {
			return "", err
		}
		if !allowMissing {
			return "", sys_error.ArticlePreviewUnavailableErr
		}
		// 失效的符号链接本身仍有目录项, 普通缺失路径则没有.
		_, statErr := os.Lstat(candidate)
		if statErr == nil {
			return "", sys_error.ArticlePreviewUnavailableErr
		}
		if !errors.Is(statErr, fs.ErrNotExist) {
			return "", statErr
		}
		missing = append(missing, filepath.Base(candidate))
		candidate = filepath.Dir(candidate)
	}
}

// imageURL 将文章引用的本地图片映射到该文件的公开 URL.
func (a *Article) imageURL(destination string) (string, error) {
	u, err := url.Parse(destination)
	if err != nil {
		return "", fmt.Errorf("%w: invalid image URI", sys_error.ArticlePreviewUnavailableErr)
	}
	if u.Host != "" && (u.Scheme == "http" || u.Scheme == "https" || u.Scheme == "") {
		return destination, nil
	}
	if u.Scheme != "" || u.Path == "" || strings.Contains(u.Path, "\\") {
		return "", sys_error.ArticlePreviewUnavailableErr
	}
	name := filepath.FromSlash(u.Path)
	if strings.HasPrefix(u.Path, "/") {
		name = strings.TrimPrefix(name, string(filepath.Separator))
	} else {
		name = filepath.Join(a.articleDir, name)
	}
	if !a.isPublicImagePath(filepath.Clean(name)) {
		return "", sys_error.ArticlePreviewUnavailableErr
	}
	actual, err := a.resolveStoragePath(name, true)
	if err != nil {
		return "", err
	}
	if !a.isPublicImagePath(actual) {
		return "", sys_error.ArticlePreviewUnavailableErr
	}
	path := filepath.ToSlash(actual)
	segments := strings.Split(path, "/")
	for i, segment := range segments {
		segments[i] = url.PathEscape(segment)
	}
	result := *a.assetBase
	result.Path = strings.TrimRight(result.Path, "/") + "/" + path
	result.RawPath = strings.TrimRight(a.assetBase.EscapedPath(), "/") + "/" + strings.Join(segments, "/")
	result.RawQuery, result.ForceQuery = u.RawQuery, u.ForceQuery
	result.Fragment, result.RawFragment = u.Fragment, u.RawFragment
	return result.String(), nil
}

// isPublicImagePath 检查 Nginx 对隐藏路径分量和 .md 的限制.
func (a *Article) isPublicImagePath(name string) bool {
	if !filepath.IsLocal(name) || strings.EqualFold(filepath.Ext(name), ".md") {
		return false
	}
	for _, segment := range strings.Split(filepath.ToSlash(name), "/") {
		if strings.HasPrefix(segment, ".") {
			return false
		}
	}
	return true
}

在存储根内读取正文、定位图片

readPreviewContent 打开配置的存储根，调用 readMarkdown 读取正文，再将绑定到当前文章的 a.imageURL 传给图片解析器。这些方法见上方的 biz/article.go 完整示例。

readMarkdown 同时识别 testUpload/.../文章.md 和历史写法 /solar-notes-data/testUpload/.../文章.md。根相对存储值没有开头的斜杠，数据库里的绝对路径必须位于配置的根目录下。

文章和图片都先解析符号链接，用 filepath.Rel 检查真实位置；实际读取文章时再用持有的 os.Root 打开，并检查文件类型。这样同名前缀目录不能冒充存储根。示例使用工程的 Go 1.25 及 Linux/macOS 支持的 O_NONBLOCK。Go 官方对 os.Root 的说明。

图片目标示例的处理方式
./img/a.png、../img/a.png从文章的真实所在目录解析。上一级仍在存储根内即可使用。
/shared/a.png以 Nginx 站点根为起点。公开地址中不会出现磁盘根目录名。
./图%201.png?v=2#demo解开输入路径的 URL 编码一次，再按文件名重新编码。保留查询参数及片段，避免把 %20 变成 %2520。
HTTP(S) 或 //cdn.example/...原样保留。API 不向该地址发请求。
普通图片暂缺根据最近存在的父目录校验根边界，仍生成图片 URL。浏览器收到该图片的 404，其余正文仍可预览。
越界、失效符号链接、隐藏路径、指向 .md 的本地图片返回 20401，避免向客户端发出错误的资源地址。示例仅为本地文件与 HTTP(S) 网络图片提供映射，其他带协议的目标按不可预览处理。

只替换图片目标，保留 Markdown 源文

article_preview/markdown.go 使用 Goldmark 定位图片语法。每遇到一个图片目标，RewriteImages 就调用传入的 a.imageURL，由当前文章解析其公开地址。在原始字节上仅替换图片目标，标题、普通链接、行内代码和围栏代码块都可保留。引用式图片与普通链接共用定义时，只把图片的那一次使用展开为行内图片，保留原定义供普通链接使用。此处处理 Markdown 图片语法，原始 HTML 标签留给前端渲染和净化。CommonMark 图片语法。

package article_preview

import (
	"bytes"
	"errors"
	"html"
	"regexp"
	"sort"
	"strings"

	"github.com/yuin/goldmark/ast"
	"github.com/yuin/goldmark/parser"
	"github.com/yuin/goldmark/text"
	"github.com/yuin/goldmark/util"
)

// imageEdit 记录一次针对源码字节区间的替换.
type imageEdit struct {
	start, end int    // start 和 end 标记左闭右开的字节范围.
	value      string // value 是新写入的 Markdown 片段.
}

// imageParser 记录 Goldmark 解析图片时消费的源码位置. 每篇文章创建一个实例.
type imageParser struct {
	parser.InlineParser                              // InlineParser 委托给 Goldmark 的标准链接解析器.
	resolve             func(string) (string, error) // resolve 将单个目标转换为公开 URL.
	edits               []imageEdit                  // edits 只收集需要改写的图片.
	err                 error                        // err 保留第一次解析或映射失败的原因.
}

// RewriteImages 重写图片目标, 并按原字节顺序拼回其余正文.
func RewriteImages(source []byte, resolve func(string) (string, error)) (string, error) {
	p := &imageParser{InlineParser: parser.NewLinkParser(), resolve: resolve}
	inlines := parser.DefaultInlineParsers()
	for i, entry := range inlines {
		if entry.Value == p.InlineParser {
			inlines[i].Value = p
		}
	}
	markdown := parser.NewParser(
		parser.WithBlockParsers(parser.DefaultBlockParsers()...),
		parser.WithInlineParsers(inlines...),
		parser.WithParagraphTransformers(parser.DefaultParagraphTransformers()...),
	)
	markdown.Parse(text.NewReader(source))
	if p.err != nil {
		return "", p.err
	}
	sort.Slice(p.edits, func(i, j int) bool { return p.edits[i].start < p.edits[j].start })
	var out strings.Builder
	cursor := 0
	for _, edit := range p.edits {
		if edit.start < cursor || edit.end < edit.start || edit.end > len(source) {
			return "", errors.New("invalid Markdown image range")
		}
		out.Write(source[cursor:edit.start])
		out.WriteString(edit.value)
		cursor = edit.end
	}
	out.Write(source[cursor:])
	return out.String(), nil
}

// CloseBlock 转发标准解析器的收尾操作, 防止链接状态跨段落残留.
func (p *imageParser) CloseBlock(parent ast.Node, r text.Reader, ctx parser.Context) {
	p.InlineParser.(parser.CloseBlocker).CloseBlock(parent, r, ctx)
}

// legacyImage 兼容简单单行图片中带裸空格的文件名. 标准标题和尖括号写法交给 Goldmark.
var legacyImage = regexp.MustCompile(`^!\[[^\[\]\\\x60\r\n]*\]\(([^()<>"'\\\t\r\n ]+(?: +[^()<>"'\\\t\r\n ]+)+)\)`)

// Parse 在解析器处理行内语法时定位图片, 记录真正需要替换的区间.
func (p *imageParser) Parse(parent ast.Node, r text.Reader, ctx parser.Context) ast.Node {
	line, start := r.PeekLine()
	if line[0] == '!' {
		if match := legacyImage.FindSubmatchIndex(line); match != nil {
			p.replace(line[match[2]:match[3]], start.Start+match[2], start.Start+match[3], nil)
			r.Advance(match[1])
			return ast.NewString(line[:match[1]])
		}
	}
	lineNo, saved := r.Position()
	node := p.InlineParser.Parse(parent, r, ctx)
	img, ok := node.(*ast.Image)
	if !ok || len(img.Destination) == 0 {
		return node
	}
	endLine, end := r.Position()
	afterLabel := start.Start + 1
	if len(line) > 1 && line[1] == '(' && end.Start > afterLabel {
		// 只定位目标, 保留括号, 标题及其间的排版.
		r.SetPosition(lineNo, saved)
		r.Advance(2)
		r.SkipSpaces()
		_, dest := r.Position()
		from := dest.Start
		if r.Peek() == '<' {
			from++
		}
		r.SetPosition(endLine, end)
		to := from + len(img.Destination)
		if from < 0 || to > len(r.Source()) || !bytes.Equal(r.Source()[from:to], img.Destination) {
			p.err = errors.New("cannot locate Markdown image destination")
			return node
		}
		p.replace(img.Destination, from, to, nil)
	} else {
		// 把这次图片引用展开为行内图片, 普通链接可以继续使用同一份引用定义.
		p.replace(img.Destination, afterLabel, end.Start, img)
	}
	return node
}

// markdownEscape 匹配单个反斜杠标点转义或带分号的字符实体.
var markdownEscape = regexp.MustCompile(`\\[[:punct:]]|&#(?:[0-9]{1,7}|[xX][0-9a-fA-F]{1,6});|&[A-Za-z][A-Za-z0-9]*;`)

// markdownValue 一次解开原文中的转义和实体, 保留 URL 百分号编码及转义后产生的字面文本.
func markdownValue(raw []byte) string {
	return markdownEscape.ReplaceAllStringFunc(string(raw), decodeMarkdownEscape)
}

// decodeMarkdownEscape 保留未知的实体名称, 数字实体按十进制或显式十六进制解码.
func decodeMarkdownEscape(token string) string {
	if token[0] == '\\' {
		return token[1:]
	}
	if token[1] == '#' {
		return html.UnescapeString(token)
	}
	if entity, ok := util.LookUpHTML5EntityByName(token[1 : len(token)-1]); ok {
		return string(entity.Characters)
	}
	return token
}

// urlSyntax 保护目标中的 Markdown 定界符和字符实体.
var urlSyntax = strings.NewReplacer("(", "%28", ")", "%29", "<", "%3C", ">", "%3E",
	" ", "%20", "\\", "%5C", "\r", "%0D", "\n", "%0A", "&", "&amp;")

// replace 仅在目标变化时记录替换, 外部 HTTP(S) 图片可完整保留原文.
func (p *imageParser) replace(raw []byte, start, end int, ref *ast.Image) {
	if p.err != nil {
		return
	}
	value := markdownValue(raw)
	target, err := p.resolve(value)
	if err != nil {
		p.err = err
		return
	}
	if target == value {
		return
	}
	target = urlSyntax.Replace(target)
	if ref != nil {
		if len(ref.Title) > 0 {
			title := html.EscapeString(markdownValue(ref.Title))
			title = strings.NewReplacer("\\", "\\\\", "\r", "&#13;", "\n", "&#10;").Replace(title)
			target += ` "` + title + `"`
		}
		target = "(" + target + ")"
	}
	p.edits = append(p.edits, imageEdit{start: start, end: end, value: target})
}

package article

import (

)

// bindPreview 读取一个 JSON 对象, 检查多余输入并执行 binding 校验.
func bindPreview(c *gin.Context, req *reqArticle.PreviewReq) error {
	decoder := json.NewDecoder(c.Request.Body)
	err := decoder.Decode(req)
	if err != nil {
		return err
	}
	err = decoder.Decode(&struct{}{})
	if err != io.EOF {
		return errors.New("request body must contain one JSON object")
	}
	return binding.Validator.ValidateStruct(req)
}

// Preview 用工程的统一包装器返回文件预览或公开的业务错误.
func Preview(c *gin.Context) {
	c.Header("Cache-Control", "no-store")
	body := &resp_body.RespBody{}
	req := &reqArticle.PreviewReq{}
	err := bindPreview(c, req)
	if err != nil {
		body.ValidateError(err)
		c.JSON(http.StatusOK, body)
		return
	}

	article := &biz.Article{}
	err = article.Preview(c.Request.Context(), req)
	if err != nil {
		var dbErr *sys_error.DatabaseError
		switch {
		case errors.Is(err, sys_error.CatalogueNotFoundErr):
			body.CatalogueNotFoundError()
		case errors.Is(err, sys_error.ArticlePreviewUnavailableErr):
			body.ArticlePreviewUnavailableError()
		case errors.As(err, &dbErr):
			body.DBError(dbErr)
		default:
			body.InternalError(err)
		}
		c.JSON(http.StatusOK, body)
		return
	}

	dto := &respArticle.PreviewResp{
		ID:      strconv.FormatInt(article.ID, 10),
		Title:   article.Title,
		Content: article.Content,
	}
	body.Success(map[string]any{"article": dto})
	c.JSON(http.StatusOK, body)
}
