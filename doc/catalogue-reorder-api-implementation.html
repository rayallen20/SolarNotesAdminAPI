POST /v1/catalogue/reorder

保存一个父目录下的阅读顺序

管理员在抽屉内调整直接子项，点击「保存排序」时提交一次写请求。后端核对打开抽屉时的分组，按目标顺序分配序号，并在同一个事务里保存对应的显隐状态。

从抽屉快照到一次保存

package catalogue

// ReorderReq 是保存同级阅读顺序的请求体.
type ReorderReq struct {
	Catalogue *ReorderCatalogueReq `json:"catalogue" binding:"required"` // Catalogue 描述所选父目录和两份分组.
}

// ReorderCatalogueReq 标识一次保存操作.
type ReorderCatalogueReq struct {
	ParentID int64        `json:"parent_id,string" binding:"gt=0"` // ParentID 为父文件夹本身的 ID.
	Expected *OrderGroups `json:"expected" binding:"required"`     // Expected 是打开排序抽屉时的分组.
	Target   *OrderGroups `json:"target" binding:"required"`       // Target 是保存后的分组.
}

// OrderGroups 要求两个数组都存在, 非 nil 的空数组合法.
type OrderGroups struct {
	SortedChildren   []ChildRef `json:"sorted_children" binding:"required,dive"`   // SortedChildren 的位置决定阅读顺序.
	UnsortedChildren []ChildRef `json:"unsorted_children" binding:"required,dive"` // UnsortedChildren 只表达成员集合.
}

// ChildRef 引用父文件夹的一个直接子节点.
type ChildRef struct {
	ID int64 `json:"id,string" binding:"gt=0"` // ID 是以十进制字符串传入的节点 ID.
}

// Reorder 校验编排请求, 使用锁定后的同级目录行比较快照并保存新顺序.
func (c *Catalogue) Reorder(ctx context.Context, req *reqCatalogue.ReorderReq) error {
	if c == nil {
		return &sys_error.InternalError{Cause: errors.New("biz.Catalogue.Reorder: nil receiver")}
	}
	if ctx == nil {
		return &sys_error.InternalError{Cause: errors.New("biz.Catalogue.Reorder: nil context")}
	}
	if req == nil {
		return &sys_error.InternalError{Cause: errors.New("biz.Catalogue.Reorder: nil request")}
	}
	plan, err := newReorderPlan(req.Catalogue)
	if err != nil {
		return err
	}
	orm := &model.Catalogue{}
	err = orm.ReorderChildren(ctx, req.Catalogue.ParentID, plan.changes)
	switch {
	case err == nil:
		return nil
	case errors.Is(err, model.CatalogueNotFoundErr):
		return sys_error.CatalogueNotFoundErr
	case errors.Is(err, model.BookNotFoundErr):
		return sys_error.BookNotFoundErr
	case errors.Is(err, model.CatalogueParentNotFolderErr):
		return fmt.Errorf("%w: parent_id must reference a folder", sys_error.CatalogueOrderInvalidErr)
	case errors.Is(err, model.CatalogueOrderChangedErr), errors.Is(err, sys_error.CatalogueOrderChangedErr):
		return sys_error.CatalogueOrderChangedErr
	default:
		return &sys_error.DatabaseError{Cause: err}
	}
}

先比请求内部，再比数据库

比较对象比较方法不一致时
expected 与 target每份分组内部两数组互斥且无重复，不含父 ID；合并后的两份 ID 集合必须完全相等。请求自身不成立，返回 10101。
锁定的直接子项与 expected已排项取当前 sort 升序的 ID 序列，待排项只比 ID 集合。忽略显隐、待排数组顺序和实际序号的间隔。返回 20302，数据库的任何目录行均不应被改写。
锁定的直接子项与 target以锁定时的状态决定最终状态；按目标数组下标加一重新连续编号。只为实际变化的行安排更新。

若快照与目标都漏掉了同一个数据库子项，两份请求的成员仍相等；与数据库比对时会得到 20302。若只有目标漏掉了该子项，两份请求已经自相矛盾，直接得到 10101。若已排节点原来的序号为 2, 7，且目标保留相同 ID 顺序，快照比对会通过，保存后序号归整为 1, 2。只有比对通过且最终序号、显隐均相同，才能跳过写入。

package biz

import (
	"fmt"
	"math"

	"SolarNotesAdminAPI/enum"
	"SolarNotesAdminAPI/model"
	reqCatalogue "SolarNotesAdminAPI/param/req/v1/catalogue"
	"SolarNotesAdminAPI/sys_error"
)

// siblingOrder 以 ID 查找请求中的位置. 0 仅在此内存索引中表示待排, 写库时转为 NULL.
type siblingOrder map[int64]uint32

// reorderPlan 保存已通过请求校验的原分组与目标分组.
type reorderPlan struct {
	expected siblingOrder // expected 保存打开抽屉时的相对位置.
	target   siblingOrder // target 保存提交后的位置, 正数从 1 开始.
}

// newReorderPlan 在获取数据库锁之前验证两份分组使用相同的完整成员集合.
func newReorderPlan(input *reqCatalogue.ReorderCatalogueReq) (*reorderPlan, error) {
	if input == nil {
		return nil, fmt.Errorf("%w: catalogue is required", sys_error.CatalogueOrderInvalidErr)
	}
	if input.ParentID <= 0 {
		return nil, fmt.Errorf("%w: parent_id must be positive", sys_error.CatalogueOrderInvalidErr)
	}
	expected, err := indexOrderGroups(input.ParentID, input.Expected, "expected")
	if err != nil {
		return nil, err
	}
	target, err := indexOrderGroups(input.ParentID, input.Target, "target")
	if err != nil {
		return nil, err
	}
	if len(expected) != len(target) {
		return nil, fmt.Errorf("%w: expected and target have different child counts", sys_error.CatalogueOrderInvalidErr)
	}
	for id := range expected {
		if _, exists := target[id]; !exists {
			return nil, fmt.Errorf("%w: target is missing child %d", sys_error.CatalogueOrderInvalidErr, id)
		}
	}
	return &reorderPlan{expected: expected, target: target}, nil
}

// indexOrderGroups 将两组 ID 合并为同一个索引, 同组和跨组重复均可被识别.
func indexOrderGroups(parentID int64, groups *reqCatalogue.OrderGroups, field string) (siblingOrder, error) {
	if groups == nil {
		return nil, fmt.Errorf("%w: %s is required", sys_error.CatalogueOrderInvalidErr, field)
	}
	if groups.SortedChildren == nil || groups.UnsortedChildren == nil {
		return nil, fmt.Errorf("%w: %s must contain both arrays", sys_error.CatalogueOrderInvalidErr, field)
	}
	if uint64(len(groups.SortedChildren)) > math.MaxUint32 {
		return nil, fmt.Errorf("%w: %s exceeds the maximum sort value", sys_error.CatalogueOrderInvalidErr, field)
	}
	order := make(siblingOrder, len(groups.SortedChildren)+len(groups.UnsortedChildren))
	for i, child := range groups.SortedChildren {
		if err := order.add(parentID, child.ID, uint32(i)+1); err != nil {
			return nil, fmt.Errorf("%s.sorted_children: %w", field, err)
		}
	}
	for _, child := range groups.UnsortedChildren {
		if err := order.add(parentID, child.ID, 0); err != nil {
			return nil, fmt.Errorf("%s.unsorted_children: %w", field, err)
		}
	}
	return order, nil
}

// add 只接受同一快照中首次出现的正数子节点 ID.
func (o siblingOrder) add(parentID, id int64, position uint32) error {
	if id <= 0 || id == parentID {
		return fmt.Errorf("%w: child ID %d must be positive and different from parent_id", sys_error.CatalogueOrderInvalidErr, id)
	}
	if _, exists := o[id]; exists {
		return fmt.Errorf("%w: duplicate child ID %d", sys_error.CatalogueOrderInvalidErr, id)
	}
	o[id] = position
	return nil
}

// matches 比对已排序列和待排集合. 输入行中非 NULL 项必须已经按 sort 升序排列.
func (o siblingOrder) matches(rows []model.Catalogue) bool {
	if len(o) != len(rows) {
		return false
	}
	var rankedCount uint64
	for _, row := range rows {
		var position uint64
		if row.Sort != nil {
			rankedCount++
			position = rankedCount
		}
		expectedPosition, exists := o[row.ID]
		if !exists || uint64(expectedPosition) != position {
			return false
		}
	}
	return true
}

// changes 根据锁定行计算目标状态, 按 ID 返回排序或显隐有变更的直接子节点.
func (p *reorderPlan) changes(rows []model.Catalogue) (map[int64]model.Catalogue, error) {
	if !p.expected.matches(rows) {
		return nil, sys_error.CatalogueOrderChangedErr
	}
	changes := make(map[int64]model.Catalogue)
	for _, row := range rows {
		target := row
		if rank := p.target[row.ID]; rank > 0 {
			target.Sort = &rank
			if row.Sort == nil {
				target.DisplayStatus = enum.DisplayVisible
			}
		} else {
			target.Sort = nil
			target.DisplayStatus = enum.DisplayHidden
		}
		if !equalCatalogueSort(row.Sort, target.Sort) || row.DisplayStatus != target.DisplayStatus {
			changes[row.ID] = target
		}
	}
	return changes, nil
}

// equalCatalogueSort 按数值比较排序位置, 两项均为 nil 表示均待排.
func equalCatalogueSort(a, b *uint32) bool {
	if a == nil || b == nil {
		return a == b
	}
	return *a == *b
}

锁定同一层，原子保存最终状态

业务层和 DAO 共用 model.Catalogue。事务锁定的行保存原状态，业务层按 ID 返回有变更的目录副本。DAO 对照原状态释放旧序号，再从副本中读取 sort 和 display_status 写入这两列。移回待排的节点在释放旧序号时已经写成 NULL / hidden，只需更新一次。

  1. 定位。 用父目录 ID 查询所属书籍。这个普通读取只用于确定第一把锁的对象。
  2. 复核。 在同一写事务内，先锁书籍，再按 ID 重新读取并锁定父目录，最后锁定该书该父目录下的全部直接子项。按 book_id、parent_id 限定范围。传入其他父目录或其他书籍的节点，都会在成员校验或事务内的快照比对阶段被拒绝。
  3. 先释放、再分配。 保留锁定行原来的序号与显隐。释放发生变化且原先非空的旧序号，再写最终的 1..N 和显隐；移回待排的子项最终同时为 NULL / hidden。交换 1、2 时，即使有 uk_book_parent_sort 唯一索引也能完成。临时 NULL 符合 chk_catalogue_sort_by_parent 对非根子节点的约束。
  4. 提交。 中间任何错误使整批回滚；只有提交成功才返回成功。所有写语句都携带本书、本父和子节点 ID 条件。

书籍行是同一本书目录写操作的共同锁。后续显隐、导入、删除或移动目录的实现也应遵循「书 → 父目录 → 子项」的顺序；这样能在一次排序事务中稳定地核对成员，并读取其他管理员刚提交的显隐修改。书级串行化涵盖同书的不同父目录，不同书籍可分别操作。

ReorderChildren 开启事务后，通过父目录实例调用私有方法锁定子节点、释放旧序号并写入目标顺序。父目录实例的 BookID 和 ID 共同限定写入范围。

// ReadBookTree 在同一个只读快照内,读取给定书籍和该书全部目录行
func (c *Catalogue) ReadBookTree(ctx context.Context, bookID int64) (*Book, []Catalogue, error) {
	if c == nil {
		err := errors.New("model.Catalogue.ReadBookTree: nil receiver")
		return nil, nil, err
	}

	if ctx == nil {
		err := errors.New("model.Catalogue.ReadBookTree: nil context")
		return nil, nil, err
	}

	if bookID <= 0 {
		err := errors.New("model.Catalogue.ReadBookTree: book ID must be positive")
		return nil, nil, err
	}

	if db.DB == nil {
		err := errors.New("model.Catalogue.ReadBookTree: database is not initialized")
		return nil, nil, err
	}

	tx := db.DB.WithContext(ctx).Begin(&sql.TxOptions{
		Isolation: sql.LevelRepeatableRead,
		ReadOnly:  true,
	})
	if tx.Error != nil {
		return nil, nil, tx.Error
	}
	defer tx.Rollback()

	var book Book
	err := tx.Select("id", "title", "sort", "display_status").
		Where("id = ?", bookID).Take(&book).Error
	if err != nil {
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return nil, nil, BookNotFoundErr
		}

		return nil, nil, err
	}

	var rows []Catalogue
	err = tx.Model(&Catalogue{}).
		Select("id", "book_id", "parent_id", "name", "intro", "type", "sort", "display_status").
		Where("book_id = ?", bookID).Find(&rows).Error
	if err != nil {
		return nil, nil, err
	}

	err = tx.Commit().Error
	if err != nil {
		return nil, nil, err
	}

	return &book, rows, nil
}

// ReorderChildren 锁定直接子节点, 由 plan 按 ID 返回有变更的目标状态, 再原子提交排序.
func (c *Catalogue) ReorderChildren(ctx context.Context, parentID int64, plan func([]Catalogue) (map[int64]Catalogue, error)) error {
	if c == nil {
		return errors.New("model.Catalogue.ReorderChildren: nil receiver")
	}
	if ctx == nil {
		return errors.New("model.Catalogue.ReorderChildren: nil context")
	}
	if parentID <= 0 {
		return errors.New("model.Catalogue.ReorderChildren: invalid parent ID")
	}
	if plan == nil {
		return errors.New("model.Catalogue.ReorderChildren: nil plan")
	}
	if db.DB == nil {
		return errors.New("database is not initialized")
	}

	conn := db.DB.WithContext(ctx)
	var parent Catalogue
	// 此次普通读只定位要锁的书. 所有决策都依据事务内的当前读.
	err := conn.Select("id", "book_id").Where("id = ?", parentID).Take(&parent).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		return CatalogueNotFoundErr
	}
	if err != nil {
		return err
	}

	tx := conn.Begin(&sql.TxOptions{Isolation: sql.LevelRepeatableRead})
	if tx.Error != nil {
		return tx.Error
	}
	defer tx.Rollback()

	rows, err := parent.lockChildren(tx)
	if err != nil {
		return err
	}
	changes, err := plan(rows)
	if err != nil {
		return err
	}
	err = parent.writeChildOrder(tx, rows, changes)
	if err != nil {
		return err
	}
	return tx.Commit().Error
}

// lockChildren 依次锁定书籍, 父节点和该书中它的全部直接子节点.
func (c *Catalogue) lockChildren(tx *gorm.DB) ([]Catalogue, error) {
	lock := clause.Locking{Strength: "UPDATE"}
	var book Book
	err := tx.Clauses(lock).Select("id").Where("id = ?", c.BookID).Take(&book).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		return nil, BookNotFoundErr
	}
	if err != nil {
		return nil, err
	}

	var parent Catalogue
	err = tx.Clauses(lock).Select("id", "book_id", "type").
		Where("id = ?", c.ID).Take(&parent).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		return nil, CatalogueNotFoundErr
	}
	if err != nil {
		return nil, err
	}
	if parent.BookID != c.BookID {
		return nil, CatalogueOrderChangedErr
	}
	if parent.Type != enum.TypeFolder {
		return nil, CatalogueParentNotFolderErr
	}

	var rows []Catalogue
	err = tx.Clauses(lock).Select("id", "book_id", "parent_id", "sort", "display_status").
		Where("book_id = ? AND parent_id = ?", c.BookID, c.ID).
		Order("sort ASC").Order("id ASC").Find(&rows).Error
	return rows, err
}

// writeChildOrder 先释放变更项占用的旧序号, 再分配最终序号, 避免同层唯一键冲突.
func (c *Catalogue) writeChildOrder(tx *gorm.DB, rows []Catalogue, changes map[int64]Catalogue) error {
	if len(changes) == 0 {
		return nil
	}
	for _, row := range rows {
		if _, changed := changes[row.ID]; !changed || row.Sort == nil {
			continue
		}
		if err := c.updateChildOrder(tx, row.ID, nil, enum.DisplayHidden); err != nil {
			return err
		}
	}
	for _, row := range rows {
		target, changed := changes[row.ID]
		if !changed {
			continue
		}
		if row.Sort != nil && target.Sort == nil {
			continue // 该项在释放旧号时已落到最终的 NULL 和 hidden.
		}
		if err := c.updateChildOrder(tx, row.ID, target.Sort, target.DisplayStatus); err != nil {
			return err
		}
	}
	return nil
}

// updateChildOrder 更新一行已锁定且实际变更的子节点, nil 会写入 SQL NULL.
func (c *Catalogue) updateChildOrder(tx *gorm.DB, childID int64, rank *uint32, status string) error {
	result := tx.Model(&Catalogue{}).
		Where("book_id = ? AND parent_id = ? AND id = ?", c.BookID, c.ID, childID).
		Updates(map[string]any{"sort": rank, "display_status": status})
	if result.Error != nil {
		return result.Error
	}
	if result.RowsAffected != 1 {
		return fmt.Errorf("catalogue %d order update affected %d rows", childID, result.RowsAffected)
	}
	return nil
}

InnoDB 的 FOR UPDATE 会读取最新可锁定的记录并持锁至事务结束。MySQL 唯一索引允许可空列有多个 NULL，因此可以在一次事务内释放旧位次，再分配最终序号。MySQL 锁定读取 · MySQL 唯一索引

联调与并发验收

在可写的隔离测试库准备两个文件夹或两本书，至少准备一个待排节点和两个已排节点。用 /v1/catalogue/tree 取测试父目录完整的 children，在 Postman 中选择 Bearer Token、raw / JSON。每次检查响应后再重读目录树，同时比对书籍、父目录和另一层子项。

操作应观察到的结果
首次将两个待排项加入，随后互换两个已排项两次均成功；顺序变为目标序列，序号连续且没有唯一键冲突。
将全部已排项移回待排；再加入其中一项先全部 null / hidden，重新加入的项为 1 / visible。空文件夹的两份分组均显式传空数组可成功。
节点已排序且手动隐藏；在抽屉中移出又加回，或只调换它的位置新序号正确，仍为 hidden。更深层子项和书籍显隐均保持各自原值。
仅改变 expected 或 target 中待排 ID 的顺序待排按集合理解，保存成功；若最终状态已经吻合，整次操作可以没有 UPDATE。
排序快照中重复、跨数组重复、包含父节点、target 少一项、数字型 ID、缺数组或 null10101，整批无写入。
用合法的父 ID，快照中含同书其他父项的孩子、别书孩子，或者两份分组均漏列一个现有孩子20302；实际层及其他层全部保持原状。
父 ID 指向文件、已删除的父项；移除令牌分别返回 10101、20301、20004。
甲、乙携带同一快照保存不同目标，甲先实际改变顺序或分组并提交乙获取书锁后读到甲的结果，返回 20302。若甲仅改变显隐，乙仍可提交，并保留持续已排节点的最新显隐。
在隔离库中让第二阶段的某次写入失败整个事务回滚，原有序号、显隐、其他层数据均保持调用前状态。

若响应超时或丢失，客户端先重读树；按已排序列、待排集合比较当前分组与原 target。若已达成目标即可结束；否则让管理员基于最新分组确认，以新的 expected 再次提交。原快照已经失效时，即使重发相同请求也可能收到 20302。
